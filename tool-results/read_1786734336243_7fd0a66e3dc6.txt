    32→// Pre-existing tolerant mock kept as a last-resort fallback when neither
    33→// strict JSON nor ES3 parsing succeed. The new ES3 parser is the primary
    34→// path; this only kicks in for totally unstructured text.
    35→function regexFallback(text: string, fileName: string): ES3ParseResult {
    36→  const detected: string[] = []
    37→  let money = 0
    38→  let fp = 0
    39→  let day = 1
    40→
    41→  const moneyMatch = text.match(/"?Funds"?\s*:\s*\{[^}]*"value"\s*:\s*([0-9.]+)/i)
    42→  if (moneyMatch) {
    43→    money = Number(moneyMatch[1])
    44→    detected.push('Funds (regex)')
    45→  }
    46→  const fpMatch = text.match(/"?FranchisePoints"?\s*:\s*\{[^}]*"value"\s*:\s*([0-9.]+)/i)
    47→  if (fpMatch) {
    48→    fp = Number(fpMatch[1])
    49→    detected.push('FranchisePoints (regex)')
    50→  }
    51→  const dayMatch = text.match(/"?Day"?\s*:\s*\{[^}]*"value"\s*:\s*([0-9]+)/i)
    52→  if (dayMatch) {
    53→    day = Number(dayMatch[1])
    54→    detected.push('Day (regex)')
    55→  }
    56→
    57→  const snapshot: SaveSnapshot = {
    58→    source: `upload:${fileName}`,
    59→    parseStatus: detected.length > 0 ? 'partial' : 'failed',
    60→    confidence: detected.length > 0 ? 'proxy' : 'unverified',
    61→    detectedFields: detected,
    62→    unknownFields: [],
    63→    money,
    64→    franchisePoints: fp,
    65→    day,
    66→    unlockedProductTiers: Array.from(new Set(ENC.products.map((p) => p.tier))).sort((a, b) => a - b),
    67→    unlockedProducts: ENC.products.map((p) => p.id),
    68→    productPlayerPricing: {},
    69→    perks: [],
    70→    extraUpgrades: [],
    71→    employees: [],
    72→    storeLayout: ENC.storeLayout,
    73→    inventoryByProduct: {},
    74→    storageInventory: {},
    75→    weather: 'unknown',
    76→    temperature: [],
    77→    roomId: null,
    78→    playerSlots: 0,
    79→    parsedAt: new Date().toISOString(),
    80→  }
    81→
    82→  return {
    83→    snapshot,
    84→    detected,
    85→    unknown: [],
    86→    confidence: snapshot.confidence,
    87→    status: snapshot.parseStatus,
    88→    typeTally: {},
    89→    warnings: ['ES3 + strict-JSON parse failed; used last-resort regex heuristics. Only scalar fields may be recovered.'],
    90→    fieldCount: detected.length,
    91→    bytesIn: text.length,
    92→    bytesOut: text.length,
    93→  }
    94→}

   113→export function Upload() {
   114→  const setSnapshot = useSaveStore((s) => s.setSnapshot)
   115→  const loadDemo = useSaveStore((s) => s.loadDemo)

   117→  const snapshot = useSaveStore((s) => s.snapshot)
   118→  const setView = useUIStore((s) => s.setView)
   119→
   120→  const [parseResult, setParseResult] = useState<ES3ParseResult | null>(null)
   121→  const [dragOver, setDragOver] = useState(false)
   122→  const [parsing, setParsing] = useState(false)

   124→  const fileInputRef = useRef<HTMLInputElement>(null)
   125→
   126→  const triggerFilePicker = useCallback(() => {
   127→    fileInputRef.current?.click()
   128→  }, [])
   129→
   130→  const runParse = useCallback(
   131→    (text: string, fileName: string) => {
   132→      let result = parseSaveFile(text, fileName)
   133→      // If both strict-JSON and ES3 paths failed, try regex fallback.
   134→      if (result.status === 'failed') {
   135→        const fb = regexFallback(text, fileName)
   136→        if (fb.detected.length > result.detected.length) result = fb
   137→      }
   138→      setParseResult(result)
   139→      if (result.status === 'failed') {
   140→        toast.error('無法解析此檔案。顯示偵測結果，但未載入。')
   141→      } else {
   142→        setSnapshot(result.snapshot)
   143→        const extras = result.snapshot.employees.length
   144→          ? `、${result.snapshot.employees.length} 員工`
   145→          : ''
   146→        toast.success(
   147→          `解析完成：${result.detected.length} 個欄位、Day ${result.snapshot.day}、$${result.snapshot.money.toFixed(0)}${extras} (${result.status})`,
   148→        )
   149→      }
   150→    },
   151→    [setSnapshot],
   152→  )
   153→
   154→  const handleFile = useCallback(
   155→    async (file: File) => {
   156→      setParsing(true)
   157→      try {
   158→        const text = await file.text()
   159→        runParse(text, file.name)
   160→      } catch (e: any) {
   161→        toast.error('解析失敗: ' + (e?.message ?? 'unknown'))
   162→        setParseResult({
   163→          snapshot: {} as SaveSnapshot,
   164→          detected: [],
   165→          unknown: [],
   166→          confidence: 'unverified',
   167→          status: 'failed',
   168→          typeTally: {},
   169→          warnings: [e?.message ?? 'unknown'],
   170→          fieldCount: 0,
   171→          bytesIn: 0,
   172→          bytesOut: 0,
   173→        })
   174→      } finally {
   175→        setParsing(false)
   176→      }
   177→    },
   178→    [runParse],
   179→  )
   180→
   181→  const handleLoadSample = useCallback(async () => {
   182→    setLoadingSample(true)
   183→    try {
   184→      const res = await fetch('/api/sample-save')
   185→      if (!res.ok) {
   186→        const j = await res.json().catch(() => ({}))
   187→        toast.error(j.message ?? `取樣本失敗 (${res.status})`)
   188→        return
   189→      }
   190→      const text = await res.text()
   191→      runParse(text, '_latest.json')
   192→    } catch (e: any) {
   193→      toast.error('取樣本失敗: ' + (e?.message ?? 'unknown'))
   194→    } finally {
   195→      setLoadingSample(false)
   196→    }
   197→  }, [runParse])
   198→
   199→  const onDrop = useCallback(
   200→    (e: React.DragEvent) => {
   201→      e.preventDefault()
   202→      setDragOver(false)
   203→      const file = e.dataTransfer.files?.[0]
   204→      if (file) handleFile(file)
   205→    },
   206→    [handleFile],
   207→  )
   208→
   209→  const s = parseResult?.snapshot
   210→  const typeEntries = parseResult ? Object.entries(parseResult.typeTally).sort((a, b) => b[1] - a[1]) : []
   211→
   212→  return (
   213→    <div className="mx-auto max-w-[1200px] space-y-4 p-4">
   214→      <SectionHeader
   215→        title="存檔上傳與本地解析"
   216→        description="支援真實 EasySave3 (.es3 / .json) 存檔。解析器會修復 ES3 內嵌值語法、解開 __type/value 信封、並將 PascalCase 欄位對應到 SaveSnapshot。所有解析在瀏覽器本地完成。"
   217→        confidence="confirmed"
   218→        note="ES3 parser 已整合 — 支援 Funds/Day/FP/Pricing/Tiers/Layout/Employees/Loan/Invoices/Upgrades 等 30+ 欄位。"
   219→      />
   220→
   221→      {/* Drop zone */}
   222→      <Card
   223→        className={`border-2 border-dashed transition-colors ${dragOver ? 'border-primary bg-primary/5' : 'border-muted'}`}
   224→        onDragOver={(e) => {
   225→          e.preventDefault()
   226→          setDragOver(true)
   227→        }}
   228→        onDragLeave={() => setDragOver(false)}
   229→        onDrop={onDrop}
   230→      >
   231→        <CardContent className="flex flex-col items-center justify-center gap-3 py-10 text-center">
   232→          <div className="flex h-12 w-12 items-center justify-center rounded-full bg-primary/10">
   233→            <FileUp className="h-6 w-6 text-primary" />
   234→          </div>
   235→          <div>
   236→            <div className="text-sm font-medium">拖曳 .es3 / .json 檔案到此，或選擇檔案</div>
   237→            <div className="mt-1 text-xs text-muted-foreground">
   238→              支援真實 EasySave3 存檔（含內嵌 bool/float/int/string 語法）、乾淨 JSON 快照、部分 JSON。
   239→              所有解析在瀏覽器本地完成，不上傳伺服器。
   240→            </div>
   241→          </div>
   242→          <div className="flex flex-wrap justify-center gap-2">
   243→            <input
   244→              ref={fileInputRef}
   245→              type="file"
   246→              accept=".es3,.json,.txt"
   247→              className="hidden"
   248→              onChange={(e) => {
   249→                const f = e.target.files?.[0]
   250→                if (f) handleFile(f)
   251→                e.target.value = ''
   252→              }}
   253→            />
   254→            <Button disabled={parsing} onClick={triggerFilePicker} className="cursor-pointer">
   255→              {parsing ? (
   256→                <>
   257→                  <Loader2 className="mr-1.5 h-4 w-4 animate-spin" /> 解析中…

   265→            <Button variant="outline" onClick={handleLoadSample} disabled={loadingSample || parsing}>
   266→              {loadingSample ? (

   268→                  <Loader2 className="mr-1.5 h-4 w-4 animate-spin" /> 載入中…
   269→                </>
   270→              ) : (

   272→                  <FlaskConical className="mr-1.5 h-4 w-4" /> 載入範本 _latest.json

   274→              )}

   277→              <Sparkles className="mr-1.5 h-4 w-4" /> 載入 Demo 存檔
   278→            </Button>
   279→            {snapshot && (
   280→              <Button variant="ghost" onClick={() => { clear(); setParseResult(null) }}>
   281→                <XCircle className="mr-1.5 h-4 w-4" /> 清除
   282→              </Button>
   283→            )}
   284→          </div>
   285→        </CardContent>
   286→      </Card>
   287→
   288→      {/* Parse status */}
   289→      {parseResult && s && (
   290→        <Card>
   291→          <CardHeader className="pb-2">
   292→            <CardTitle className="flex items-center justify-between text-base">
   293→              <span className="flex items-center gap-2">
   294→                {parseResult.status === 'ok' ? (
   295→                  <CheckCircle2 className="h-5 w-5 text-emerald-500" />
   296→                ) : parseResult.status === 'partial' ? (
   297→                  <AlertTriangle className="h-5 w-5 text-amber-500" />
   298→                ) : (
   299→                  <XCircle className="h-5 w-5 text-rose-500" />
   300→                )}
   301→                解析狀態: {parseResult.status.toUpperCase()}
   302→                <Badge variant="outline" className="ml-1 text-[10px]">
   303→                  {parseResult.fieldCount} 欄位
   304→                </Badge>
   305→                <Badge variant="outline" className="text-[10px]">
   306→                  {(parseResult.bytesIn / 1024).toFixed(1)} KB → {(parseResult.bytesOut / 1024).toFixed(1)} KB
   307→                </Badge>
   308→              </span>
   309→              <ConfidenceBadge confidence={parseResult.confidence} />
   310→            </CardTitle>
   311→          </CardHeader>
   312→          <CardContent className="space-y-4">
   313→            {/* Key scalars */}
   314→            <div className="grid grid-cols-2 gap-3 sm:grid-cols-4 lg:grid-cols-6">
   315→              <StatCard label="Money" value={s.money > 0 ? fmtMoney(s.money) : '—'} confidence="confirmed" />
   316→              <StatCard label="Day" value={String(s.day)} confidence="confirmed" />
   317→              <StatCard label="Franchise Pts" value={String(s.franchisePoints)} confidence="confirmed" />
