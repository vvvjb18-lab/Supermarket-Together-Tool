    66→    66→// ============================================================
    67→    67→// Demand proxy
    68→    68→// ------------------------------------------------------------
    69→    69→// For every customer type, take necessitiesChances[necessityIndex].
    70→    70→// For each necessity pool, distribute weight over RAW TOKENS (preserving duplicates).
    71→    71→// Product demand proxy = Σ (weight / rawPoolLength) for every token hit.
    72→    72→//
    73→    73→// This is a PROXY: true customer spawn distribution and runtime product
    74→    74→// selection method are not fully extracted. We assume:
    75→    75→//  - equal customer-type spawn (configurable in simulator)
    76→    76→//  - uniform pick within a necessity's raw token pool
    77→    77→//  - one purchase per customer (maxProductsCustomersToBuy=5 in tuning, but
    78→    78→//    for a single-necessity proxy we count one hit per chosen necessity)
    79→    79→// ============================================================
    80→    80→
    81→    81→export interface DemandOptions {
    82→    82→  customerWeights?: number[] // per-customer-type spawn weight (default: equal = 1)
    83→    83→  mode?: 'raw' | 'unique' // raw preserves duplicate tokens; unique dedupes
    84→    84→}
    85→    85→
    86→    86→export function computeDemandProxy(
    87→    87→  productId: number,
    88→    88→  necessities: Necessity[],
    89→    89→  customerTypes: CustomerType[],
    90→    90→  opts: DemandOptions = {},
    91→    91→): CalcResult<number> {
    92→    92→  const mode = opts.mode ?? 'raw'
    93→    93→  const cw = opts.customerWeights
    94→    94→  let total = 0
    95→    95→  const hits: string[] = []
    96→    96→  for (let ci = 0; ci < customerTypes.length; ci++) {
    97→    97→    const cust = customerTypes[ci]
    98→    98→    const w = cw ? cw[ci] ?? 0 : 1
    99→    99→    if (w <= 0) continue
   100→   100→    const chances = cust.necessitiesChances
   101→   101→    for (let ni = 0; ni < necessities.length; ni++) {
   102→   102→      const chance = chances[ni] ?? 0
   103→   103→      if (chance <= 0) continue
   104→   104→      const nec = necessities[ni]
   105→   105→      const tokens = mode === 'unique' ? Array.from(new Set(nec.rawTokens)) : nec.rawTokens
   106→   106→      if (tokens.length === 0) continue
   107→   107→      const hitsCount = tokens.filter((t) => t === productId).length
   108→   108→      if (hitsCount === 0) continue
   109→   109→      // contribution = customerWeight × necessityChance × (tokenHits / poolSize)
   110→   110→      const contribution = w * chance * (hitsCount / tokens.length)
   111→   111→      total += contribution
   112→   112→      hits.push(`cust[${ci}]×${w} × nec[${ni}]×${chance} × ${hitsCount}/${tokens.length}`)
   113→   113→    }
   114→   114→  }
   115→   115→  // normalize by sum of customer weights so it reads as "expected purchases per customer"
   116→   116→  const weightSum = cw ? cw.reduce((a, b) => a + b, 0) : customerTypes.length
   117→   117→  const normalized = weightSum > 0 ? total / weightSum : 0
   118→   118→  return {
   119→   119→    value: normalized,
   120→   120→    formula:
   121→   121→      mode === 'raw'
   122→   122→        ? 'demandProxy = Σcust Σnec (custWeight × necChance × tokenHits/rawPoolSize) / ΣcustWeight'
   123→   123→        : 'demandProxy (unique mode) = Σcust Σnec (custWeight × necChance × 1/uniquePoolSize if product present) / ΣcustWeight',
   124→   124→    sources: hits.slice(0, 6),
   125→   125→    confidence: 'proxy',
   126→   126→    note: 'Assumes equal customer spawn + uniform pick within necessity pool. True runtime selection unverified.',
   127→   127→  }
   128→   128→}

   138→   138→    confidence: 'proxy',
   139→   139→  }
   140→   140→}

   175→   175→export function computeSaltProbe(): SaltProbeResult {
   176→   176→  const nec = ENC.necessities
   177→   177→  const custs = ENC.customerTypes
   178→   178→  const salt = ENC.products.find((p) => p.id === 4)!
   179→   179→  const nec9 = nec[9] // Salt single product, rawIds "4-4-4-4-4"
   180→   180→  const nec10 = nec[10] // Staple Groceries, 62 raw tokens
   181→   181→
   182→   182→  // Route via necessity[9]: only customer #47 has weight 0.5
   183→   183→  let route9 = 0
   184→   184→  const route9Hits: string[] = []
   185→   185→  for (let ci = 0; ci < custs.length; ci++) {
   186→   186→    const w = 1 // equal spawn
   187→   187→    const chance = custs[ci].necessitiesChances[9] ?? 0
   188→   188→    if (chance <= 0) continue
   189→   189→    // raw pool = [4,4,4,4,4], length 5, salt hits 5/5 = 1.0
   190→   190→    const hits = nec9.rawTokens.filter((t) => t === 4).length
   191→   191→    const contribution = (w * chance * hits) / nec9.rawTokens.length
   192→   192→    route9 += contribution
   193→   193→    route9Hits.push(`cust[${ci}] weight=${w} × nec[9] chance=${chance} × ${hits}/${nec9.rawTokens.length}`)
   194→   194→  }
   195→   195→  route9 = route9 / custs.length // normalize by equal-spawn sum
   196→   196→
   197→   197→  // Route via necessity[10]: staple, 62 tokens, salt appears once
   198→   198→  let route10 = 0
   199→   199→  const route10Hits: string[] = []
   200→   200→  for (let ci = 0; ci < custs.length; ci++) {
   201→   201→    const w = 1
   202→   202→    const chance = custs[ci].necessitiesChances[10] ?? 0
   203→   203→    if (chance <= 0) continue
   204→   204→    const hits = nec10.rawTokens.filter((t) => t === 4).length // =1
   205→   205→    const contribution = (w * chance * hits) / nec10.rawTokens.length
   206→   206→    route10 += contribution
   207→   207→    route10Hits.push(`cust[${ci}] × nec[10] chance=${chance} × ${hits}/${nec10.rawTokens.length}`)
   208→   208→  }
   209→   209→  route10 = route10 / custs.length
   210→   210→
   211→   211→  const total = route9 + route10
   212→   212→
   213→   213→  // comparison: pick a sample of early products + premium electronics
   214→   214→  const compareIds = [0, 1, 2, 3, 4, 5, 9, 11, 48, 54, 108, 173, 175, 287, 296, 299]
   215→   215→  const comparison = compareIds
   216→   216→    .map((id) => ENC.products.find((p) => p.id === id))
   217→   217→    .filter((p): p is Product => !!p)
   218→   218→    .map((p) => {
   219→   219→      const demand = computeDemandProxy(p.id, nec, custs).value
   220→   220→      const vol = computeColliderVolume(p).value
   221→   221→      const box = computeBoxValue(p).value
   222→   222→      return {
   223→   223→        product: p,
   224→   224→        basePrice: p.basePricePerUnit,
   225→   225→        boxValue: box,
   226→   226→        volume: vol,
   227→   227→        demandProxy: demand,
   228→   228→        shelfEfficiency: vol > 0 ? (demand * box) / vol : 0,
   229→   229→      }
   230→   230→    })
   231→   231→    .sort((a, b) => b.demandProxy - a.demandProxy)
   232→   232→
   233→   233→  return {
   234→   234→    saltProduct: salt,
   235→   235→    saltRoute9: {
   236→   236→      value: route9,
   237→   237→      formula: 'Σcust (custWeight × nec[9].chance × 5/5) / 58  — only cust#47 has chance=0.5',
   238→   238→      sources: route9Hits,
   239→   239→      confidence: 'confirmed',
   240→   240→      note: 'necessity[9] rawIds="4-4-4-4-4", salt hits 5/5 tokens',
   241→   241→    },
   242→   242→    saltRoute10: {
   243→   243→      value: route10,
   244→   244→      formula: 'Σcust (custWeight × nec[10].chance × 1/62) / 58  — salt appears once in 62-token staple pool',
   245→   245→      sources: route10Hits,
   246→   246→      confidence: 'confirmed',
   247→   247→      note: 'uniform random assumption → ~1.61% hit per staple-weighted customer',
   248→   248→    },
   249→   249→    saltTotalDemand: {
   250→   250→      value: total,
   251→   251→      formula: 'saltTotalDemand = route9 + route10',
   252→   252→      sources: [`route9=${route9.toFixed(6)}`, `route10=${route10.toFixed(6)}`],
   253→   253→      confidence: 'proxy',
   254→   254→      note: 'sum of two confirmed routes; proxy because spawn distribution unverified',
   255→   255→    },
   256→   256→    saltBoxValue: computeBoxValue(salt),
   257→   257→    saltVolume: computeColliderVolume(salt),
   258→   258→    saltValueDensity: computeValueDensity(salt),
   259→   259→    comparison,
   260→   260→    conclusion:
   261→   261→      'Salt 是已確認的特殊機制（necessity[9] 單一商品、rawIds="4-4-4-4-4"），但不是已確認的賺錢 exploit。它有壟斷路徑（route 9），但只有 customer #47 觸發、且單價極低（$0.45）。透過 Staple Groceries 路徑只佔 1/62 ≈ 1.61% 機率。整體 demand proxy 偏低，單位利潤受限。建議實測而非盲信。',
   262→   262→    confidence: 'exploit',
   263→   263→  }
   264→   264→}

   270→   270→export interface NecessityCoverage {
   271→   271→  necessityIndex: number
   272→   272→  necessityName: string
   273→   273→  totalWeight: number // sum of customer chances for this necessity
   274→   274→  stockedTokens: number // raw tokens covered by stocked products
   275→   275→  totalTokens: number
   276→   276→  coverage: number // 0..1
   277→   277→  missingProducts: number[]
   278→   278→}
   279→   279→
   280→   280→export function computeNecessityCoverage(
   281→   281→  stockedProductIds: Set<number>,
   282→   282→  necessities: Necessity[] = ENC.necessities,
   283→   283→  customerTypes: CustomerType[] = ENC.customerTypes,
   284→   284→): CalcResult<NecessityCoverage[]> {
   285→   285→  const coverage: NecessityCoverage[] = necessities.map((nec) => {
   286→   286→    const totalWeight = customerTypes.reduce((s, c) => s + (c.necessitiesChances[nec.index] ?? 0), 0)
   287→   287→    const stockedTokens = nec.rawTokens.filter((t) => t >= 0 && stockedProductIds.has(t)).length
   288→   288→    const missing = Array.from(new Set(nec.rawTokens.filter((t) => t >= 0 && !stockedProductIds.has(t))))
   289→   289→    return {
   290→   290→      necessityIndex: nec.index,
   291→   291→      necessityName: nec.name.en,
   292→   292→      totalWeight,
   293→   293→      stockedTokens,
   294→   294→      totalTokens: nec.rawTokens.length,
   295→   295→      coverage: nec.rawTokens.length > 0 ? stockedTokens / nec.rawTokens.length : 0,
   296→   296→      missingProducts: missing,
   297→   297→    }
   298→   298→  })
   299→   299→  return {
   300→   300→    value: coverage,
   301→   301→    formula: 'coverage = stockedRawTokens / totalRawTokens  (per necessity, weighted by Σ customer chances)',
   302→   302→    sources: [`necessities=${necessities.length}`, `customerTypes=${customerTypes.length}`, `stockedProducts=${stockedProductIds.size}`],
   303→   303→    confidence: 'proxy',
   304→   304→    note: 'Raw token mode preserves duplicates (Salt counts 5×).',
   305→   305→  }
   306→   306→}

   309→   309→// Shelf efficiency (per layout prop)
   310→   310→// ============================================================

   312→   312→export interface PropEfficiency {
   313→   313→  propIndex: number
   314→   314→  buildableId: number
   315→   315→  totalUnits: number
   316→   316→  distinctProducts: number
   317→   317→  shelfValue: number // sum of count × basePrice
   318→   318→  demandCoverage: number // sum of count × demandProxy (proxy)
   319→   319→  emptySlots: number // inventory entries with count 0 or product -1
   320→   320→  negativeAnomalies: number
   321→   321→  duplicatedProducts: number // products appearing on this prop AND another prop
   322→   322→  efficiency: number // shelfValue × demandCoverage normalized
   323→   323→}
   324→   324→
   325→   325→export function computeShelfEfficiency(
   326→   326→  layout: LayoutProp[],
   327→   327→  products: Product[] = ENC.products,
   328→   328→  necessities: Necessity[] = ENC.necessities,
   329→   329→  customerTypes: CustomerType[] = ENC.customerTypes,
   330→   330→): CalcResult<PropEfficiency[]> {
   331→   331→  const productById = new Map(products.map((p) => [p.id, p]))
   332→   332→  // precompute demand proxies
   333→   333→  const demandCache = new Map<number, number>()
   334→   334→  for (const p of products) {
   335→   335→    demandCache.set(p.id, computeDemandProxy(p.id, necessities, customerTypes).value)
   336→   336→  }
   337→   337→  // count product appearances across props
   338→   338→  const propAppearances = new Map<number, number>()
   339→   339→  for (const prop of layout) {
   340→   340→    for (const inv of prop.inventory) {
   341→   341→      if (inv.product < 0) continue
   342→   342→      propAppearances.set(inv.product, (propAppearances.get(inv.product) ?? 0) + 1)
   343→   343→    }
   344→   344→  }
   345→   345→  const result: PropEfficiency[] = layout.map((prop) => {
   346→   346→    let totalUnits = 0
   347→   347→    let shelfValue = 0
   348→   348→    let demandCoverage = 0
   349→   349→    let emptySlots = 0
   350→   350→    let negativeAnomalies = 0
   351→   351→    let duplicated = 0
   352→   352→    const seen = new Set<number>()
   353→   353→    for (const inv of prop.inventory) {
   354→   354→      if (inv.product < 0 || inv.count === 0) {
   355→   355→        emptySlots++
   356→   356→        continue
   357→   357→      }
   358→   358→      if (inv.count < 0) {
   359→   359→        negativeAnomalies++
   360→   360→        continue
   361→   361→      }
   362→   362→      totalUnits += inv.count
   363→   363→      const p = productById.get(inv.product)
   364→   364→      if (p) {
   365→   365→        shelfValue += inv.count * p.basePricePerUnit
   366→   366→        demandCoverage += inv.count * (demandCache.get(inv.product) ?? 0)
   367→   367→      }
   368→   368→      if (seen.has(inv.product)) duplicated++
   369→   369→      else seen.add(inv.product)
   370→   370→      if ((propAppearances.get(inv.product) ?? 0) > 1) duplicated++
   371→   371→    }
   372→   372→    const efficiency = shelfValue * (1 + demandCoverage * 100)
   373→   373→    return {
   374→   374→      propIndex: prop.index,
   375→   375→      buildableId: prop.buildableId,
   376→   376→      totalUnits,
   377→   377→      distinctProducts: seen.size,
   378→   378→      shelfValue,
   379→   379→      demandCoverage,
   380→   380→      emptySlots,
   381→   381→      negativeAnomalies,
   382→   382→      duplicatedProducts: duplicated,
   383→   383→      efficiency,
   384→   384→    }
   385→   385→  })
   386→   386→  return {
   387→   387→    value: result,
   388→   388→    formula: 'shelfValue = Σ count×basePrice; demandCoverage = Σ count×demandProxy; efficiency = shelfValue×(1+demandCoverage×100)',
   389→   389→    sources: [`layout props=${layout.length}`, `products=${products.length}`],
   390→   390→    confidence: 'proxy',
   391→   391→  }
   392→   392→}

   405→   405→
   406→   406→export function computeDashboardScores(snapshot: SaveSnapshot | null): DashboardScores {
   407→   407→  // demand coverage
   408→   408→  let demandCov = 0
   409→   409→  if (snapshot && Object.keys(snapshot.inventoryByProduct).length > 0) {
   410→   410→    const stocked = new Set(
   411→   411→      Object.entries(snapshot.inventoryByProduct)
   412→   412→        .filter(([, c]) => (c as number) > 0)
   413→   413→        .map(([id]) => Number(id)),
   414→   414→    )
   415→   415→    const cov = computeNecessityCoverage(stocked).value
   416→   416→    const totalWeight = cov.reduce((s, c) => s + c.totalWeight, 0)
   417→   417→    const coveredWeight = cov.reduce((s, c) => s + c.coverage * c.totalWeight, 0)
   418→   418→    demandCov = totalWeight > 0 ? (coveredWeight / totalWeight) * 100 : 0
   419→   419→  } else {
   420→   420→    // demo: assume all products stocked
   421→   421→    demandCov = 100
   422→   422→  }
   423→   423→
   424→   424→  // stock risk: % of stocked products with very low units (<5)
   425→   425→  let stockRisk = 30
   426→   426→  if (snapshot && Object.keys(snapshot.inventoryByProduct).length > 0) {
   427→   427→    const entries = Object.entries(snapshot.inventoryByProduct)
   428→   428→    const low = entries.filter(([, c]) => (c as number) > 0 && (c as number) < 5).length
   429→   429→    const total = entries.filter(([, c]) => (c as number) > 0).length
   430→   430→    stockRisk = total > 0 ? (low / total) * 100 : 0
   431→   431→  }
   432→   432→
   433→   433→  // shelf efficiency
   434→   434→  let shelfEff = 60
   435→   435→  if (snapshot && snapshot.storeLayout.length > 0) {
   436→   436→    const eff = computeShelfEfficiency(snapshot.storeLayout).value
   437→   437→    const emptyTotal = eff.reduce((s, p) => s + p.emptySlots, 0)
   438→   438→    const slotTotal = eff.reduce((s, p) => s + (p.emptySlots + p.distinctProducts), 0)
   439→   439→    shelfEff = slotTotal > 0 ? Math.max(0, 100 - (emptyTotal / slotTotal) * 100) : 0
   440→   440→  }
   441→   441→
   442→   442→  // employee efficiency (demo: based on employee count & avg level)
   443→   443→  let empEff = 50
   444→   444→  if (snapshot && snapshot.employees.length > 0) {
   445→   445→    const avgLevel = snapshot.employees.reduce((s, e) => {
   446→   446→      const levels = Object.values(e.skills).map((sk) => sk.level)
   447→   447→      return s + (levels.reduce((a, b) => a + b, 0) / (levels.length || 1))
   448→   448→    }, 0) / snapshot.employees.length
   449→   449→    empEff = Math.min(100, 30 + avgLevel * 14)
   450→   450→  }
   451→   451→
   452→   452→  const storeHealth = Math.max(0, Math.min(100, demandCov * 0.35 + (100 - stockRisk) * 0.25 + shelfEff * 0.25 + empEff * 0.15))
   453→   453→
   454→   454→  return {
   455→   455→    storeHealth: {
   456→   456→      value: storeHealth,
   457→   457→      formula: 'storeHealth = 0.35×demandCov + 0.25×(100-stockRisk) + 0.25×shelfEff + 0.15×empEff',
   458→   458→      sources: [],
   459→   459→      confidence: 'proxy',
   460→   460→    },
   461→   461→    demandCoverage: { value: demandCov, formula: 'Σ(nec.coverage × nec.totalWeight) / Σ nec.totalWeight × 100', sources: [], confidence: 'proxy' },
   462→   462→    stockRisk: { value: stockRisk, formula: '(lowStockProducts / totalStockedProducts) × 100, low=<5 units', sources: [], confidence: 'proxy' },
   463→   463→    shelfEfficiency: { value: shelfEff, formula: '100 − (emptySlots/totalSlots)×100', sources: [], confidence: 'proxy' },
   464→   464→    employeeEfficiency: { value: empEff, formula: 'min(100, 30 + avgLevel×14)', sources: [], confidence: 'proxy' },
   465→   465→  }
   466→   466→}

   473→   473→  | 'balanced'

   481→   481→export interface RestockRecommendation {
   482→   482→  productId: number
   483→   483→  productName: string
   484→   484→  buyBoxes: number
   485→   485→  units: number
   486→   486→  costEstimate: number
   487→   487→  revenueProxy: number
   488→   488→  reason: string
   489→   489→  priority: number
   490→   490→}

   492→   492→export function computeRestockPlan(
   493→   493→  snapshot: SaveSnapshot | null,
   494→   494→  strategy: RestockStrategy,
   495→   495→  budget: number,
   496→   496→  options: { season?: number } = {},
   497→   497→): CalcResult<RestockRecommendation[]> {
   498→   498→  const products = ENC.products
   499→   499→  const necessities = ENC.necessities
   500→   500→  const custs = ENC.customerTypes
   501→   501→  const inv = snapshot?.inventoryByProduct ?? {}
   502→   502→  const unlocked = snapshot?.unlockedProducts ?? products.map((p) => p.id)
   503→   503→  const unlockedSet = new Set(unlocked)
   504→   504→
   505→   505→  // score each product
   506→   506→  const scored = products
   507→   507→    .filter((p) => unlockedSet.has(p.id))
   508→   508→    .map((p) => {
   509→   509→      const demand = computeDemandProxy(p.id, necessities, custs).value
   510→   510→      const box = computeBoxValue(p).value
   511→   511→      const vol = computeColliderVolume(p).value
   512→   512→      const density = computeValueDensity(p).value
   513→   513→      const current = (inv[p.id] as number) ?? 0
   514→   514→      const isPremium = ENC.premiumProducts.includes(p.id)
   515→   515→      const isSeasonal = options.season != null && ENC.seasons[options.season]?.productIds.includes(p.id)
   516→   516→      let score = 0
   517→   517→      let reason = ''
   518→   518→      switch (strategy) {
   519→   519→        case 'demand-coverage':
   520→   520→          score = demand
   521→   521→          reason = `demand proxy ${demand.toFixed(5)}; current ${current} units`
   522→   522→          break
   523→   523→        case 'high-profit':
   524→   524→          score = demand * box
   525→   525→          reason = `weighted box proxy ${(demand * box).toFixed(3)}`
   526→   526→          break
   527→   527→        case 'high-density':
   528→   528→          score = density
   529→   529→          reason = `value density ${density.toFixed(2)} $/unit³`
   530→   530→          break
   531→   531→        case 'premium-push':
   532→   532→          score = (isPremium ? 2 : 0.1) * box
   533→   533→          reason = isPremium ? 'premium product, high box value' : 'filler'
   534→   534→          break
   535→   535→        case 'seasonal-prep':
   536→   536→          score = (isSeasonal ? 3 : 0.05) * demand * box
   537→   537→          reason = isSeasonal ? `in-season (${ENC.seasons[options.season!].name.en})` : 'off-season filler'
   538→   538→          break
   539→   539→        case 'early-game-cheap-fill':
   540→   540→          score = p.tier <= 5 ? (1 / (p.basePricePerUnit + 0.1)) * (demand + 0.01) : -1
   541→   541→          reason = `tier ${p.tier} cheap fill, base $${p.basePricePerUnit}`
   542→   542→          break
   543→   543→        case 'balanced':
   544→   544→        default:
   545→   545→          score = demand * box * 0.5 + density * 0.2 + (current < 10 ? 5 : 0)
   546→   546→          reason = `balanced: demand×box ${(demand * box).toFixed(3)}, density ${density.toFixed(1)}, stock ${current}`
   547→   547→          break
   548→   548→      }
   549→   549→      // urgency boost if low stock & has demand
   550→   550→      if (current < 5 && demand > 0.001) {
   551→   551→        score *= 1.5
   552→   552→        reason += '; LOW STOCK urgency ×1.5'
   553→   553→      }
   554→   554→      return { p, demand, box, vol, current, score, reason }
   555→   555→    })
   556→   556→    .filter((s) => s.score > 0)
   557→   557→    .sort((a, b) => b.score - a.score)
   558→   558→
   559→   559→  // greedy knapsack: take boxes while budget remains
   560→   560→  const recs: RestockRecommendation[] = []
   561→   561→  let remaining = budget
   562→   562→  for (const s of scored) {
   563→   563→    if (remaining <= 0) break
   564→   564→    const boxCost = s.box // buying one box costs ≈ boxValue (wholesale ≈ base × units)
   565→   565→    if (boxCost <= 0) continue
   566→   566→    const maxBoxesByBudget = Math.floor(remaining / boxCost)
   567→   567→    const desiredBoxes = Math.max(1, Math.min(maxBoxesByBudget, Math.ceil((50 - s.current) / s.p.maxItemsPerBox)))
   568→   568→    if (desiredBoxes <= 0) continue
   569→   569→    const boxes = Math.max(1, Math.min(desiredBoxes, maxBoxesByBudget))
   570→   570→    if (boxes <= 0) continue
   571→   571→    const cost = boxes * boxCost
   572→   572→    const units = boxes * s.p.maxItemsPerBox
   573→   573→    const revenue = s.demand * units * s.p.basePricePerUnit
   574→   574→    recs.push({
   575→   575→      productId: s.p.id,
   576→   576→      productName: s.p.name.en,
   577→   577→      buyBoxes: boxes,
   578→   578→      units,
   579→   579→      costEstimate: cost,
   580→   580→      revenueProxy: revenue,
   581→   581→      reason: s.reason,
   582→   582→      priority: recs.length + 1,
   583→   583→    })
   584→   584→    remaining -= cost
   585→   585→  }
   586→   586→  return {
   587→   587→    value: recs,
   588→   588→    formula: `greedy knapsack: score by strategy=${strategy}, take boxes while budget≥0; boxCost≈boxValue`,
   589→   589→    sources: [`budget=${budget}`, `candidates=${scored.length}`, `strategy=${strategy}`],
   590→   590→    confidence: 'proxy',
   591→   591→    note: 'boxCost assumes wholesale ≈ base×units; real wholesale price unverified.',
   592→   592→  }
   593→   593→}

   817→   817→    switch (strategy) {
   818→   818→      case 'employee-automation':
   819→   819→        weight = tags.includes('speed') ? 3 : tags.includes('headcount') ? 2 : 0.5
   820→   820→        break
   821→   821→      case 'checkout-speed':
   822→   822→        weight = tags.includes('throughput') ? 3 : 0.5
   823→   823→        break
   824→   824→      case 'customer-volume':
   825→   825→        weight = category === 'customer-volume' ? 3 : 0.5
   826→   826→        break
   827→   827→      case 'manufacturing':
   828→   828→        weight = category === 'ordering' ? 2 : 0.7
   829→   829→        break
   830→   830→      case 'anti-theft':
   831→   831→        weight = eff.includes('surveillance') ? 3 : 0.3
   832→   832→        break
   833→   833→      case 'early-cash':
   834→   834→        weight = tags.includes('finance') ? 2.5 : tags.includes('sales-cap') ? 2 : 0.6
   835→   835→        break
   836→   836→      case 'late-scale':
   837→   837→        weight = tags.includes('headcount') ? 2.5 : tags.includes('speed') ? 2 : 0.7
   838→   838→        break
   839→   839→    }
   840→   840→    return {
   841→   841→      skill: s,
   842→   842→      category,
   843→   843→      roiProxy: roi * weight,
   844→   844→      synergyTags: tags,
   845→   845→      confidence: s.description.en ? 'proxy' : 'unverified',
   846→   846→      note: s.effect || '(no effect extracted — possibly reserved slot)',
   847→   847→    }
   848→   848→  })
   849→   849→  rois.sort((a, b) => b.roiProxy - a.roiProxy)
   850→   850→  return {
   851→   851→    value: rois,
   852→   852→    formula: `roiProxy = baseScore × strategyWeight; strategy=${strategy}; all perks cost 1000 FP`,
   853→   853→    sources: [`skills=${ENC.skills.length}`, `strategy=${strategy}`],

   855→   855→    note: 'Exact % values for some skills not fully extracted from IL; effect text shown verbatim.',

   923→   923→export interface SimulationConfig {
   924→   924→  n: number
   925→   925→  customerWeights?: number[]
   926→   926→  mode: 'raw' | 'unique'
   927→   927→  stockedProductIds?: Set<number>

   929→   929→
   930→   930→export interface SimulationResult {
   931→   931→  productHits: Map<number, number>
   932→   932→  missedSales: Map<number, number>
   933→   933→  totalCustomers: number
   934→   934→  totalHits: number
   935→   935→  missedHits: number
   936→   936→  demandCoverage: number
   937→   937→  topMissing: { productId: number; name: string; missed: number }[]
   938→   938→  topOverstockedLowDemand: { productId: number; name: string; units: number; demand: number }[]
   939→   939→}
   940→   940→
   941→   941→export function simulateCustomers(cfg: SimulationConfig): CalcResult<SimulationResult> {
   942→   942→  const nec = ENC.necessities
   943→   943→  const custs = ENC.customerTypes
   944→   944→  const products = ENC.products
   945→   945→  const productById = new Map(products.map((p) => [p.id, p]))
   946→   946→  const cw = cfg.customerWeights ?? custs.map(() => 1)
   947→   947→  const weightSum = cw.reduce((a, b) => a + b, 0)
   948→   948→  const stocked = cfg.stockedProductIds ?? new Set<number>(products.map((p) => p.id))
   949→   949→
   950→   950→  const productHits = new Map<number, number>()
   951→   951→  const missedSales = new Map<number, number>()
   952→   952→  let totalHits = 0
   953→   953→  let missedHits = 0
   954→   954→
   955→   955→  for (let i = 0; i < cfg.n; i++) {
   956→   956→    // pick customer type by weight
   957→   957→    let r = Math.random() * weightSum
   958→   958→    let ci = 0
   959→   959→    for (; ci < custs.length; ci++) {
   960→   960→      r -= cw[ci]
   961→   961→      if (r <= 0) break
   962→   962→    }
   963→   963→    if (ci >= custs.length) ci = custs.length - 1
   964→   964→    const cust = custs[ci]
   965→   965→    // pick necessity by chance (weighted)
   966→   966→    const chances = cust.necessitiesChances
   967→   967→    const chanceSum = chances.reduce((a, b) => a + Math.max(0, b), 0)
   968→   968→    if (chanceSum <= 0) continue
   969→   969→    let r2 = Math.random() * chanceSum
   970→   970→    let ni = -1
   971→   971→    for (let k = 0; k < chances.length; k++) {
   972→   972→      r2 -= Math.max(0, chances[k])
   973→   973→      if (r2 <= 0) {
   974→   974→        ni = k
   975→   975→        break
   976→   976→      }
   977→   977→    }
   978→   978→    if (ni < 0) ni = chances.length - 1
   979→   979→    const pool = nec[ni]
   980→   980→    if (!pool || pool.rawTokens.length === 0) continue
   981→   981→    const tokens = cfg.mode === 'unique' ? Array.from(new Set(pool.rawTokens)) : pool.rawTokens
   982→   982→    const chosen = tokens[Math.floor(Math.random() * tokens.length)]
   983→   983→    if (chosen < 0) continue
   984→   984→    totalHits++
   985→   985→    if (stocked.has(chosen)) {
   986→   986→      productHits.set(chosen, (productHits.get(chosen) ?? 0) + 1)
   987→   987→    } else {
   988→   988→      missedSales.set(chosen, (missedSales.get(chosen) ?? 0) + 1)
   989→   989→      missedHits++
   990→   990→    }
   991→   991→  }
   992→   992→
   993→   993→  const demandCoverage = totalHits > 0 ? (totalHits - missedHits) / totalHits : 0
   994→   994→
   995→   995→  const topMissing = Array.from(missedSales.entries())
   996→   996→    .map(([pid, missed]) => ({ productId: pid, name: productById.get(pid)?.name.en ?? `#${pid}`, missed }))
   997→   997→    .sort((a, b) => b.missed - a.missed)
   998→   998→    .slice(0, 10)
   999→   999→
  1000→  1000→  // overstocked low demand: products with high inventory but low demand proxy
  1001→  1001→  const topOverstockedLowDemand = products
  1002→  1002→    .map((p) => {
  1003→  1003→      const demand = computeDemandProxy(p.id, nec, custs, { mode: cfg.mode }).value
  1004→  1004→      return { productId: p.id, name: p.name.en, units: 0, demand }
  1005→  1005→    })
  1006→  1006→    .filter((x) => x.demand < 0.0005)
  1007→  1007→    .slice(0, 10)
  1008→  1008→
  1009→  1009→  return {
  1010→  1010→    value: { productHits, missedSales, totalCustomers: cfg.n, totalHits, missedHits, demandCoverage, topMissing, topOverstockedLowDemand },
  1011→  1011→    formula: `Monte Carlo: pick cust by weight, pick necessity by chance, pick product uniform from ${cfg.mode} pool`,
  1012→  1012→    sources: [`n=${cfg.n}`, `mode=${cfg.mode}`, `customerWeights=${cw.length}`],
  1013→  1013→    confidence: 'proxy',
  1014→  1014→    note: 'Random uniform assumptions; true runtime selection unverified. Re-run for variance.',
  1015→  1015→  }
  1016→  1016→}
