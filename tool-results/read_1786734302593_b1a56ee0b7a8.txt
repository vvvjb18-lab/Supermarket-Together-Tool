    34→    34→function parseCustomWeights(text: string): number[] | null {
    35→    35→  const lines = text
    36→    36→    .split(/[\n,]+/)
    37→    37→    .map((l) => l.trim())
    38→    38→    .filter((l) => l.length > 0)
    39→    39→  if (lines.length !== ENC.customerTypes.length) return null
    40→    40→  const nums = lines.map(Number)
    41→    41→  if (nums.some((n) => !Number.isFinite(n) || n < 0)) return null
    42→    42→  if (nums.every((n) => n === 0)) return null
    43→    43→  return nums
    44→    44→}
    45→    45→
    46→    46→function heatColor(weight: number): string {
    47→    47→  if (weight <= 0) return 'transparent'
    48→    48→  // 0..1 → emerald with alpha
    49→    49→  const alpha = Math.min(1, Math.max(0.08, weight))
    50→    50→  return `rgba(16, 185, 129, ${alpha.toFixed(3)})`
    51→    51→}
