    71→function downloadBlob(filename: string, content: string, mime: string) {
    72→  const blob = new Blob([content], { type: mime })
    73→  const url = URL.createObjectURL(blob)
    74→  const a = document.createElement('a')
    75→  a.href = url
    76→  a.download = filename
    77→  document.body.appendChild(a)
    78→  a.click()
    79→  document.body.removeChild(a)
    80→  setTimeout(() => URL.revokeObjectURL(url), 1000)
    81→}

    83→function buildMarkdown(
    84→  strategy: RestockStrategy,
    85→  budget: number,
    86→  recs: RestockRecommendation[],
    87→  formula: string,
    88→  snapshotLabel: string,
    89→  note?: string,
    90→): string {
    91→  const totalCost = recs.reduce((s, r) => s + r.costEstimate, 0)
    92→  const totalUnits = recs.reduce((s, r) => s + r.units, 0)
    93→  const totalRev = recs.reduce((s, r) => s + r.revenueProxy, 0)
    94→  const roi = totalCost > 0 ? (totalRev / totalCost).toFixed(2) : '—'
    95→  let md = '# Supermarket Together — 補貨計畫\n\n'
    96→  md += `生成時間: ${new Date().toISOString()}\n`
    97→  md += `存檔來源: ${snapshotLabel}\n`
    98→  md += `策略: ${strategy}\n`
    99→  md += `預算: $${budget}\n\n`
   100→  md += `## 統計\n\n- 總成本: $${totalCost.toFixed(2)}\n- 總單位: ${totalUnits}\n- 營收 Proxy: $${totalRev.toFixed(2)}\n- ROI Proxy: ${roi}×\n\n`
   101→  md += `## 公式與信心\n\n\`${formula}\`\n\n`
   102→  if (note) md += `> ${note}\n\n`
   103→  md += `## 採購清單 (${recs.length} 項)\n\n`
   104→  md += `| # | Product | ID | Boxes | Units | Cost | Revenue Proxy | Reason |\n|---|---|---|---|---|---|---|---|\n`
   105→  recs.forEach((r, i) => {
   106→    md += `| ${i + 1} | ${r.productName} | ${r.productId} | ${r.buyBoxes} | ${r.units} | $${r.costEstimate.toFixed(2)} | $${r.revenueProxy.toFixed(2)} | ${r.reason} |\n`
   107→  })
   108→  return md
   109→}
   110→
   111→function buildJson(recs: RestockRecommendation[], strategy: RestockStrategy, budget: number) {
   112→  return JSON.stringify({ strategy, budget, generatedAt: new Date().toISOString(), recommendations: recs }, null, 2)
   113→}

   606→                        </tr>

   609→                  </tbody>
   610→                  <tfoot className="sticky bottom-0 bg-muted/80 font-medium">
   611→                    <tr>
   612→                      <td colSpan={2} className="p-2">合計</td>
   613→                      <td className="p-2 text-right font-mono tabular-nums">{recs.reduce((s, r) => s + r.buyBoxes, 0)}</td>
   614→                      <td className="p-2 text-right font-mono tabular-nums">{totalUnits.toLocaleString()}</td>
   615→                      <td className="p-2 text-right font-mono tabular-nums">{fmtMoney(totalCost)}</td>
   616→                      <td className="p-2 text-right font-mono tabular-nums text-emerald-600">{fmtMoney(totalRev)}</td>
   617→                      <td className="p-2 text-[10px] text-muted-foreground">ROI {fmt(roi, 2)}×</td>
   618→                      {room && <td className="p-2" />}
   619→                    </tr>
   620→                  </tfoot>
   621→                </table>
   622→              </div>
   623→            </CardContent>
   624→          </Card>
   625→        </>
   626→      )}
   627→    </div>
   628→  )
   629→}

   635→function DetectionStat({
   636→  id,
   637→  label,
   638→  sub,
   639→  count,
   640→  accent,
   641→  expanded,
   642→  setExpanded,
   643→}: {
   644→  id: string
   645→  label: string
   646→  sub: string
   647→  count: number
   648→  accent: 'warn' | 'bad' | 'neutral'
   649→  expanded: string | null
   650→  setExpanded: (s: string | null) => void
   651→}) {
   652→  const accentClass =
   653→    accent === 'bad' ? 'border-rose-500/40 text-rose-600'
   654→    : accent === 'warn' ? 'border-amber-500/40 text-amber-600'
   655→    : 'border-zinc-400/40 text-zinc-600'
   656→  const isExpanded = expanded === id
   657→  return (
   658→    <button
   659→      onClick={() => setExpanded(isExpanded ? null : id)}
   660→      className={cn(
   661→        'rounded-md border bg-card p-3 text-left transition-colors hover:bg-accent',
   662→        accentClass,
   663→        isExpanded && 'ring-1 ring-primary',
   664→      )}
   665→    >
   666→      <div className="flex items-center justify-between">
   667→        <span className="text-[11px] font-medium uppercase tracking-wide">{label}</span>
   668→        {count > 0 && <AlertTriangle className="h-3 w-3" />}
   669→      </div>
   670→      <div className="mt-1 text-2xl font-bold tabular-nums">{count}</div>
   671→      <div className="mt-0.5 text-[10px] opacity-80">{sub}</div>
   672→    </button>
   673→  )
   674→}
