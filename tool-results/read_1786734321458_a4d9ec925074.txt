     1→'use client'
     2→
     3→import { useMemo, useState } from 'react'
     4→import {
     5→  Bar,
     6→  BarChart,
     7→  CartesianGrid,
     8→  ResponsiveContainer,
     9→  Tooltip as RTooltip,
    10→  XAxis,
    11→  YAxis,
    12→} from 'recharts'
    13→import { encyclopedia as ENC, productById, productZhName } from '@/lib/data-loader'
    14→import { useRoomStore } from '@/lib/store'
    15→import {

    18→  StatCard,
    19→  MiniBar,
    20→  fmt,
    21→} from '@/components/shared/primitives'
    22→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    23→import { Badge } from '@/components/ui/badge'
    24→import { Button } from '@/components/ui/button'
    25→import {
    26→  Select,
    27→  SelectContent,
    28→  SelectItem,
    29→  SelectTrigger,
    30→  SelectValue,
    31→} from '@/components/ui/select'
    32→import {
    33→  Tooltip,
    34→  TooltipContent,
    35→  TooltipProvider,
    36→  TooltipTrigger,
    37→} from '@/components/ui/tooltip'
    38→import {
    39→  Plus,
    40→  Trash2,
    41→  Layers,
    42→  AlertTriangle,
    43→  Factory,
    44→  Boxes,
    45→  TrendingUp,
    46→} from 'lucide-react'
    47→import { cn } from '@/lib/utils'
    48→import type { ManufacturingProduct } from '@/lib/types'
    49→
    50→const PRODUCTION_BASE_TIME = 30 // seconds, from config.manufacturing.productionBaseTimeSeconds
    51→
    52→interface QueueItem {
    53→  productId: number
    54→  quantity: number
    55→}
    56→
    57→export function Manufacturing() {
    58→  const [queue, setQueue] = useState<QueueItem[]>([
    59→    { productId: 10, quantity: 5 }, // Muffin
    60→    { productId: 29, quantity: 3 }, // Smoothies
    61→  ])
    62→  const [addPick, setAddPick] = useState<string>('')
    63→
    64→  const products = ENC.manufacturingProducts
    65→
    66→  // Compute volume + density for each manufacturing product
    67→  const enriched = useMemo(() => {
    68→    return products.map((m) => {
    69→      const vol = m.size.x * m.size.y * m.size.z
    70→      const density = vol > 0 ? m.itemsPerBox / vol : 0
    71→      const linkedProduct = productById.get(m.linkedProductID)
    72→      return { m, vol, density, linkedProduct }
    73→    })
    74→  }, [products])
    75→
    76→  const densityRanking = useMemo(
    77→    () => [...enriched].sort((a, b) => b.density - a.density),
    78→    [enriched],
    79→  )
    80→
    81→  // Queue calcs
    82→  const queueEnriched = useMemo(
    83→    () =>
    84→      queue.map((q) => {
    85→        const m = products.find((p) => p.id === q.productId)!
    86→        const timeSeconds = q.quantity * PRODUCTION_BASE_TIME
    87→        const totalUnits = q.quantity * m.itemsPerBox
    88→        return { ...q, m, timeSeconds, totalUnits }
    89→      }),
    90→    [queue, products],
    91→  )
    92→
    93→  const totalSeconds = queueEnriched.reduce((a, q) => a + q.timeSeconds, 0)
    94→  const totalUnits = queueEnriched.reduce((a, q) => a + q.totalUnits, 0)
    95→  const totalBoxes = queueEnriched.reduce((a, q) => a + q.quantity, 0)
    96→
    97→  // Throughput: units per hour. assume one machine running 30s/product → 2 products/min → 120 products/hr
    98→  // then × itemsPerBox = units/hr. If parallel machines N, scale by N.
    99→  const [machines, setMachines] = useState(1)
   100→  const throughputPerHour = useMemo(() => {
   101→    if (totalSeconds <= 0) return 0
   102→    // units produced per hour at N machines:
   103→    // (3600 / 30) * itemsPerBox summed weighted = totalUnits * (3600 / totalSeconds) per machine
   104→    // with N machines, throughput scales linearly (proxy)
   105→    const perMachine = totalSeconds > 0 ? totalUnits * (3600 / totalSeconds) : 0
   106→    return perMachine * machines
   107→  }, [totalSeconds, totalUnits, machines])
   108→
   109→  // Room
   110→  const room = useRoomStore((s) => s.room)
   111→  const selfId = useRoomStore((s) => s.selfId)
   112→  const assignTask = useRoomStore((s) => s.assignTask)
   113→  const manufacturingTask = room?.tasks.find((t) => t.category === 'manufacturing')
   114→  const members = room?.members ?? []
   115→  const assignedMember = manufacturingTask
   116→    ? room?.members.find((m) => m.id === manufacturingTask.playerId)
   117→    : undefined
   118→
   119→  function addToQueue() {
   120→    if (!addPick) return
   121→    const pid = Number(addPick)
   122→    setQueue((q) => {
   123→      const ex = q.find((x) => x.productId === pid)
   124→      if (ex) return q.map((x) => (x.productId === pid ? { ...x, quantity: x.quantity + 1 } : x))
   125→      return [...q, { productId: pid, quantity: 1 }]
   126→    })
   127→    setAddPick('')
   128→  }
   129→
   130→  function updateQty(pid: number, qty: number) {
   131→    setQueue((q) => q.map((x) => (x.productId === pid ? { ...x, quantity: Math.max(0, qty) } : x)))
   132→  }
   133→
   134→  function removeItem(pid: number) {
   135→    setQueue((q) => q.filter((x) => x.productId !== pid))
   136→  }
   137→
   138→  return (
   139→    <TooltipProvider delayDuration={200}>
   140→      <div className="mx-auto max-w-[1600px] space-y-4 p-4">
   141→        {/* Header */}
   142→        <div>
   143→          <h1 className="text-2xl font-bold tracking-tight">製造實驗室</h1>
   144→          <p className="text-sm text-muted-foreground">
   145→            30 個製造產品 · 基底時間 30s/件 · 自由配方組合制 · 產能與密度推導
   146→          </p>
   147→        </div>
   148→
   149→        {/* Honesty callout */}
   150→        <div className="flex items-start gap-2 rounded-lg border border-amber-500/30 bg-amber-500/5 p-3">
   151→          <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-amber-600" />
   152→          <div className="text-xs text-amber-800 dark:text-amber-200">
   153→            <span className="font-semibold">誠實聲明：</span>
   154→            固定配方盈利未完全提取；食譜為<span className="font-semibold">基底商品 + 任意可組合一般商品</span>，無固定配方表。
   155→            本頁只做<span className="font-semibold">產能與密度推導</span>，不預測成品售價或利潤。
   156→          </div>
   157→        </div>
   158→
   159→        {/* Top stats */}
   160→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   161→          <StatCard
   162→            label="製造品總數"
   163→            value={products.length}
   164→            unit="items"
   165→            confidence="confirmed"
   166→            formula="encyclopedia.manufacturingProducts.length"
   167→            accent="neutral"
   168→          />
   169→          <StatCard
   170→            label="基底生產時間"
   171→            value={`${PRODUCTION_BASE_TIME}s`}
   172→            unit="/件"
   173→            confidence="confirmed"
   174→            formula="config.manufacturing.productionBaseTimeSeconds = 30"
   175→            accent="good"
   176→          />
   177→          <StatCard
   178→            label="佇列總時間"
   179→            value={fmtTime(totalSeconds)}
   180→            confidence="confirmed"
   181→            formula={`Σ(quantity × ${PRODUCTION_BASE_TIME}s) = ${totalSeconds}s`}
   182→            hint={`${totalBoxes} 箱 · ${totalUnits} 單位`}
   183→            accent="neutral"
   184→          />
   185→          <StatCard
   186→            label="預估吞吐量"
   187→            value={fmt(throughputPerHour, 0)}
   188→            unit="u/hr"
   189→            confidence="proxy"
   190→            formula={`throughput = totalUnits × (3600/totalSec) × ${machines} 機台`}
   191→            hint="假設並行線性擴展"
   192→            accent="warn"
   193→          />
   194→        </div>
   195→
   196→        {/* Production queue planner */}
   197→        <Card>
   198→          <CardHeader className="pb-2">
   199→            <CardTitle className="flex items-center justify-between text-base">
   200→              <span className="flex items-center gap-2">
   201→                <Factory className="h-4 w-4 text-emerald-500" />
   202→                生產佇列規劃
   203→              </span>
   204→              <div className="flex items-center gap-2">
   205→                <ConfidenceBadge confidence="confirmed" formula={`每件 ${PRODUCTION_BASE_TIME}s`} />
   206→                <ConfidenceBadge confidence="proxy" formula="throughput × machines" />
   207→              </div>
   208→            </CardTitle>
   209→          </CardHeader>
   210→          <CardContent className="space-y-3">
   211→            {/* Add row */}
   212→            <div className="flex flex-wrap items-center gap-2">
   213→              <Select value={addPick} onValueChange={setAddPick}>
   214→                <SelectTrigger className="h-8 w-[240px] text-xs">
   215→                  <SelectValue placeholder="選擇製造品加入佇列…" />
   216→                </SelectTrigger>
   217→                <SelectContent>
   218→                  {products.map((m) => (
   219→                    <SelectItem key={m.id} value={String(m.id)} className="text-xs">
   220→                      #{m.id} {m.name.zhHant || m.name.en} · {m.itemsPerBox}/box
   221→                    </SelectItem>
   222→                  ))}
   223→                </SelectContent>
   224→              </Select>
   225→              <Button size="sm" className="h-8" onClick={addToQueue} disabled={!addPick}>
   226→                <Plus className="mr-1 h-3.5 w-3.5" /> 加入
   227→              </Button>
   228→              <div className="ml-auto flex items-center gap-2">
   229→                <span className="text-xs text-muted-foreground">並行機台</span>
   230→                <Select value={String(machines)} onValueChange={(v) => setMachines(Number(v))}>
   231→                  <SelectTrigger className="h-8 w-[80px] text-xs">
   232→                    <SelectValue />
   233→                  </SelectTrigger>
   234→                  <SelectContent>
   235→                    {[1, 2, 3, 4, 5, 6, 8, 10].map((n) => (
   236→                      <SelectItem key={n} value={String(n)} className="text-xs">
   237→                        {n}×
   238→                      </SelectItem>
   239→                    ))}
   240→                  </SelectContent>
   241→                </Select>
   242→              </div>
   243→            </div>
   244→
   245→            {/* Queue table */}
   246→            {queueEnriched.length === 0 ? (
   247→              <div className="rounded-md border border-dashed bg-muted/30 py-6 text-center text-sm text-muted-foreground">
   248→                佇列為空 — 從上方選擇製造品加入
   249→              </div>
   250→            ) : (
   251→              <div className="overflow-x-auto">
   252→                <table className="w-full text-sm">
   253→                  <thead>
   254→                    <tr className="border-b text-left text-[11px] uppercase text-muted-foreground">
   255→                      <th className="py-2 pr-3 font-medium">產品</th>
   256→                      <th className="py-2 pr-3 font-medium">數量（箱）</th>
   257→                      <th className="py-2 pr-3 font-medium">items/box</th>
   258→                      <th className="py-2 pr-3 font-medium">單位</th>
   259→                      <th className="py-2 pr-3 font-medium">耗時</th>
   260→                      <th className="py-2 font-medium"></th>
   261→                    </tr>
   262→                  </thead>
   263→                  <tbody>
   264→                    {queueEnriched.map((q) => (
   265→                      <tr key={q.productId} className="border-b last:border-0">
   266→                        <td className="py-2 pr-3">
   267→                          <div className="font-medium">
   268→                            {q.m.name.zhHant || q.m.name.en}
   269→                          </div>
   270→                          <code className="text-[10px] text-muted-foreground">#{q.m.id} · {q.m.name.en}</code>
   271→                        </td>
   272→                        <td className="py-2 pr-3">
   273→                          <div className="flex items-center gap-1">
   274→                            <Button
   275→                              size="sm"
   276→                              variant="outline"
   277→                              className="h-6 w-6 p-0"
   278→                              onClick={() => updateQty(q.productId, q.quantity - 1)}
   279→                            >
   280→                              −
   281→                            </Button>
   282→                            <span className="w-8 text-center font-mono tabular-nums">{q.quantity}</span>
   283→                            <Button
   284→                              size="sm"
   285→                              variant="outline"
   286→                              className="h-6 w-6 p-0"
   287→                              onClick={() => updateQty(q.productId, q.quantity + 1)}
   288→                            >
   289→                              +
   290→                            </Button>
   291→                          </div>
   292→                        </td>
   293→                        <td className="py-2 pr-3 font-mono tabular-nums">{q.m.itemsPerBox}</td>
   294→                        <td className="py-2 pr-3 font-mono tabular-nums">{q.totalUnits}</td>
   295→                        <td className="py-2 pr-3 font-mono tabular-nums">{fmtTime(q.timeSeconds)}</td>
   296→                        <td className="py-2 text-right">
   297→                          <Button
   298→                            size="sm"
   299→                            variant="ghost"
   300→                            className="h-6 w-6 p-0 text-rose-500 hover:bg-rose-500/10"
   301→                            onClick={() => removeItem(q.productId)}
   302→                          >
   303→                            <Trash2 className="h-3.5 w-3.5" />
   304→                          </Button>
   305→                        </td>
   306→                      </tr>
   307→                    ))}
   308→                  </tbody>
   309→                  <tfoot>
   310→                    <tr className="border-t-2 bg-muted/40 font-semibold">
   311→                      <td className="py-2 pr-3">合計</td>
   312→                      <td className="py-2 pr-3 font-mono tabular-nums">{totalBoxes}</td>
   313→                      <td className="py-2 pr-3"></td>
   314→                      <td className="py-2 pr-3 font-mono tabular-nums">{totalUnits}</td>
   315→                      <td className="py-2 pr-3 font-mono tabular-nums">{fmtTime(totalSeconds)}</td>
   316→                      <td></td>
   317→                    </tr>
   318→                  </tfoot>
   319→                </table>
   320→              </div>
   321→            )}
   322→
   323→            {/* Throughput callout */}
   324→            <div className="rounded-md border bg-muted/40 p-3 text-xs">
   325→              <div className="font-mono">
   326→                throughput = {totalUnits} units × (3600 / {totalSeconds}s) × {machines} machines
   327→                {' → '}
   328→                <span className="font-bold text-emerald-600 dark:text-emerald-400">
   329→                  {fmt(throughputPerHour, 0)} units/hour
   330→                </span>
   331→              </div>
   332→              <div className="mt-1 text-muted-foreground">
   333→                proxy：假設機台並行線性擴展、無 setup/中斷。真實遊戲中可能受 factory slot 限制。
   334→              </div>
   335→            </div>
   336→          </CardContent>
   337→        </Card>
   338→
   339→        {/* Room task assignment: 製造佇列管理 */}
   340→        <Card>
   341→          <CardHeader className="pb-2">
   342→            <CardTitle className="flex items-center justify-between text-base">
   343→              <span className="flex items-center gap-2">
   344→                <Boxes className="h-4 w-4 text-sky-500" />
   345→                製造佇列管理（房間任務）
   346→              </span>
   347→              <ConfidenceBadge
   348→                confidence={room ? 'confirmed' : 'unverified'}
   349→                formula={room ? 'room.tasks[manufacturing].assignedTo' : 'no room'}
   350→                note={room ? '同步給房間' : '需建立房間'}
   351→              />
   352→            </CardTitle>
   353→          </CardHeader>
   354→          <CardContent>
   355→            {room && manufacturingTask ? (
   356→              <div className="flex flex-wrap items-center gap-3">
   357→                <div className="flex-1">
   358→                  <div className="text-sm font-medium">{manufacturingTask.label}</div>
   359→                  <div className="text-[11px] text-muted-foreground">
   360→                    task id: <code className="font-mono">{manufacturingTask.id}</code>
   361→                    {assignedMember && (
   362→                      <span className="ml-2 text-emerald-600 dark:text-emerald-400">
   363→                        · 已指派給 {assignedMember.name}
   364→                      </span>
   365→                    )}
   366→                  </div>
   367→                </div>
   368→                <Select
   369→                  value={manufacturingTask.playerId ?? ''}
   370→                  onValueChange={(v) => assignTask(manufacturingTask.id, v)}
   371→                >
   372→                  <SelectTrigger className="h-8 w-[200px] text-xs">
   373→                    <SelectValue placeholder="指派房間成員…" />
   374→                  </SelectTrigger>
   375→                  <SelectContent>
   376→                    <SelectItem value="" className="text-xs">— 未指派 —</SelectItem>
   377→                    {members.map((m) => (
   378→                      <SelectItem key={m.id} value={m.id} className="text-xs">
   379→                        <span className="flex items-center gap-1">
   380→                          <span className="inline-block h-2 w-2 rounded-full" style={{ backgroundColor: m.color }} />
   381→                          {m.name} {m.id === selfId && '(我)'}
   382→                        </span>
   383→                      </SelectItem>
   384→                    ))}
   385→                  </SelectContent>
   386→                </Select>
   387→              </div>
   388→            ) : (
   389→              <div className="rounded-md border border-dashed bg-muted/30 px-3 py-4 text-center text-xs text-muted-foreground">
   390→                未連接房間 — 此功能需先建立/加入房間後啟用。
   391→                <br />
   392→                <span className="text-[10px]">房間建立後，可將「製造佇列管理」任務指派給任一房間成員。</span>
   393→              </div>
   394→            )}
   395→          </CardContent>
   396→        </Card>
   397→
   398→        {/* Density ranking leaderboard */}
   399→        <Card>
   400→          <CardHeader className="pb-2">
   401→            <CardTitle className="flex items-center justify-between text-base">
   402→              <span className="flex items-center gap-2">
   403→                <TrendingUp className="h-4 w-4 text-fuchsia-500" />
   404→                貨架密度排行（itemsPerBox / volume）
   405→              </span>
   406→              <ConfidenceBadge
   407→                confidence="confirmed"
   408→                formula="density = itemsPerBox / (size.x × y × z)"
   409→                sources={['manufacturing.size (IL)', 'manufacturing.itemsPerBox']}
   410→                note="密度高者優先上架"
   411→              />
   412→            </CardTitle>
   413→          </CardHeader>
   414→          <CardContent>
   415→            <div className="grid gap-2 md:grid-cols-2 xl:grid-cols-3">
   416→              {densityRanking.slice(0, 12).map((row, i) => (
   417→                <DensityRow key={row.m.id} rank={i + 1} row={row} max={densityRanking[0]?.density ?? 1} />
   418→              ))}
   419→            </div>
   420→            <p className="mt-2 text-[11px] text-muted-foreground">
   421→              顯示前 12 名。完整 30 名見下方產品表。
   422→            </p>
   423→          </CardContent>
   424→        </Card>
   425→
   426→        {/* Density chart top 15 */}
   427→        <Card>
   428→          <CardHeader className="pb-2">
   429→            <CardTitle className="flex items-center justify-between text-base">
   430→              <span className="flex items-center gap-2">
   431→                <Layers className="h-4 w-4 text-emerald-500" />
   432→                密度條圖（Top 15）
   433→              </span>
   434→              <ConfidenceBadge confidence="confirmed" formula="itemsPerBox / volume" />
   435→            </CardTitle>
   436→          </CardHeader>
   437→          <CardContent>
   438→            <div className="h-[380px] w-full">
   439→              <ResponsiveContainer width="100%" height="100%">
   440→                <BarChart
   441→                  data={densityRanking.slice(0, 15)}
   442→                  layout="vertical"
   443→                  margin={{ top: 4, right: 16, bottom: 4, left: 8 }}
   444→                >
   445→                  <CartesianGrid strokeDasharray="3 3" horizontal={false} opacity={0.3} />
   446→                  <XAxis type="number" tick={{ fontSize: 10 }} />
   447→                  <YAxis
   448→                    type="category"
   449→                    dataKey={(d: typeof densityRanking[number]) => d.m.name.zhHant || d.m.name.en}
   450→                    width={120}
   451→                    tick={{ fontSize: 10 }}
   452→                  />
   453→                  <RTooltip
   454→                    cursor={{ fill: 'rgba(0,0,0,0.04)' }}
   455→                    content={({ active, payload }) => {
   456→                      if (!active || !payload || !payload.length) return null
   457→                      const r = payload[0].payload as (typeof densityRanking)[number]
   458→                      return (
   459→                        <div className="rounded-md border bg-background p-2 text-xs shadow-md">
   460→                          <div className="font-semibold">{r.m.name.zhHant || r.m.name.en}</div>
   461→                          <div className="text-muted-foreground">#{r.m.id} · {r.m.name.en}</div>
   462→                          <div className="mt-1 font-mono">
   463→                            density: <span className="font-bold">{fmt(r.density, 2)}</span> items/u³
   464→                          </div>
   465→                          <div className="font-mono text-[10px] text-muted-foreground">
   466→                            itemsPerBox={r.m.itemsPerBox} · vol={fmt(r.vol, 4)} u³
   467→                          </div>
   468→                        </div>
   469→                      )
   470→                    }}
   471→                  />
   472→                  <Bar dataKey="density" radius={[0, 4, 4, 0]} fill="#10b981" />
   473→                </BarChart>
   474→              </ResponsiveContainer>
   475→            </div>
   476→          </CardContent>
   477→        </Card>
   478→
   479→        {/* Required input products / recipe note */}
   480→        <SectionHeader
   481→          title="基底商品（linkedProductID）"
   482→          description="每個製造品有一個 linkedProductID 指向基底商品。配方為基底+任意可組合一般商品，無固定配方表。"
   483→          confidence="unverified"
   484→          formula="linkedProductID = 基底商品（productsData）"
   485→          note="食譜 = 基底 + 任意 combinable 一般商品（未提取固定表）"
   486→        />
   487→        <Card>
   488→          <CardContent className="pt-2">
   489→            <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   490→              {enriched.map(({ m, linkedProduct }) => (
   491→                <div
   492→                  key={m.id}
   493→                  className="flex items-center justify-between gap-2 rounded-md border bg-card px-3 py-2 text-sm"
   494→                >
   495→                  <div className="min-w-0">
   496→                    <div className="truncate font-medium">
   497→                      {m.name.zhHant || m.name.en}
   498→                    </div>

   640→function fmtTime(seconds: number): string {
   641→  if (seconds <= 0) return '0s'
   642→  const m = Math.floor(seconds / 60)
   643→  const s = seconds % 60
   644→  if (m === 0) return `${s}s`
   645→  return `${m}m ${s}s`
   646→}
