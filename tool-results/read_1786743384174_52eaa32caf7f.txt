    54→    54→/**
    55→    55→ * Normalized set of unlocked perk indices from a save snapshot.
    56→    56→ *
    57→    57→ * The save's `skillUnlocks` field is already an array of perk indices

    61→    61→export function getUnlockedSkillIndices(snapshot: SaveSnapshot | null): number[] {
    62→    62→  if (!snapshot?.skillUnlocks) return []
    63→    63→  // Defensive: filter to valid integers and dedupe.
    64→    64→  const seen = new Set<number>()
    65→    65→  const out: number[] = []
    66→    66→  for (const idx of snapshot.skillUnlocks) {
    67→    67→    if (typeof idx === 'number' && Number.isFinite(idx) && !seen.has(idx)) {
    68→    68→      seen.add(idx)
    69→    69→      out.push(idx)
    70→    70→    }
    71→    71→  }
    72→    72→  return out
    73→    73→}

   122→   122→export function groupSkillsByCategory(skills: Skill[]): Map<string, Skill[]> {

   137→   137→/** Generate a stable color for a category name (used for badges / dots). */
   138→   138→export function categoryColor(name: string): string {
   139→   139→  let hash = 0
   140→   140→  for (let i = 0; i < name.length; i++) {
   141→   141→    hash = (hash << 5) - hash + name.charCodeAt(i)
   142→   142→    hash |= 0
   143→   143→  }
   144→   144→  const hue = Math.abs(hash) % 360
   145→   145→  return `hsl(${hue} 65% 45%)`
   146→   146→}

   171→   171→ *   electricFactor = 0.8                    → 電費           -20%
   172→   172→ *   autopayInvoices = 1                     → 自動付款       enabled

   296→   296→/** Parse an effect string into structured metrics. */
   297→   297→export function parseEffectForMetric(effect: string): ParsedEffect[] {
   298→   298→  if (!effect || !effect.trim()) return []
   299→   299→  const out: ParsedEffect[] = []
   300→   300→  const seen = new Set<string>()
   301→   301→  for (const { re, metric, fmt } of EFFECT_PATTERNS) {
   302→   302→    const m = effect.match(re)
   303→   303→    if (m) {
   304→   304→      const key = metric + '|' + fmt(m)
   305→   305→      if (seen.has(key)) continue
   306→   306→      seen.add(key)
   307→   307→      out.push({ metric, delta: fmt(m), raw: m[0] })
   308→   308→    }
   309→   309→  }
   310→   310→  // If nothing matched and effect text is non-trivial, return a single
   311→   311→  // "未提取" entry so the UI can show something.
   312→   312→  if (out.length === 0 && effect.length > 0 && !effect.includes('no matching perk')) {
   313→   313→    out.push({ metric: '效果', delta: '未提取', raw: effect })
   314→   314→  }
   315→   315→  return out
   316→   316→}

   319→   319→// estimateSkillImpact — used by the Benefit Comparator (Tool 4)
   320→   320→// ============================================================
   321→   321→
   322→   322→export interface SkillImpact {
   323→   323→  metric: string
   324→   324→  value: string
   325→   325→  confidence: Confidence
   326→   326→}
   327→   327→
   328→   328→/**
   329→   329→ * Estimate the numerical impact of a skill, derived from its effect string.
   330→   330→ *
   331→   331→ * Confidence policy:
   332→   332→ *   - 'confirmed' — pattern matched and the effect text is direct IL code.
   333→   333→ *   - 'proxy'     — pattern matched but value is approximate (e.g. we
   334→   334→ *                   inferred -20% from electricFactor=0.8).
   335→   335→ *   - 'unverified'— no pattern matched (returned as '效果: 未提取').
   336→   336→ */
   337→   337→export function estimateSkillImpact(
   338→   338→  skill: Skill,
   339→   339→  _snapshot?: SaveSnapshot | null,
   340→   340→): SkillImpact[] {
   341→   341→  void _snapshot // currently unused; reserved for future context-aware estimates
   342→   342→  const parsed = parseEffectForMetric(skill.effect)
   343→   343→  if (parsed.length === 0) {
   344→   344→    return [{ metric: '效果', value: '未提取', confidence: 'unverified' }]
   345→   345→  }
   346→   346→  return parsed.map((p) => {
   347→   347→    // 'enabled' / '未提取' are soft — mark proxy; numeric deltas are confirmed
   348→   348→    // because they come straight from the IL effect string.
   349→   349→    const conf: Confidence =
   350→   350→      p.delta === 'enabled' || p.delta === '未提取' ? 'proxy' : 'confirmed'
   351→   351→    return { metric: p.metric, value: p.delta, confidence: conf }
   352→   352→  })
   353→   353→}

   392→   392→export function categorizeSkill(skill: Skill): SkillCategorization {
   393→   393→  const eff = skill.effect.toLowerCase()
   394→   394→  let category = 'other'
   395→   395→  if (eff.includes('speed')) category = 'employee-speed'
   396→   396→  else if (eff.includes('checkout') || eff.includes('selfcheckout') || eff.includes('productcheckout'))
   397→   397→    category = 'checkout'
   398→   398→  else if (eff.includes('customer') || eff.includes('extracustomers') || eff.includes('bystander'))
   399→   399→    category = 'customer-volume'
   400→   400→  else if (eff.includes('recycle')) category = 'recycling'
   401→   401→  else if (eff.includes('maxemployees') || eff.includes('employee')) category = 'employee-count'
   402→   402→  else if (eff.includes('manufacturing') || eff.includes('reroll') || eff.includes('ordering'))
   403→   403→    category = 'ordering'
   404→   404→  else if (eff.includes('debt') || eff.includes('electric') || eff.includes('autopay'))
   405→   405→    category = 'finance'
   406→   406→
   407→   407→  const tags: string[] = []
   408→   408→  if (eff.includes('extraemployeespeedfactor')) tags.push('speed', 'automation')
   409→   409→  if (eff.includes('maxemployees')) tags.push('headcount')
   410→   410→  if (
   411→   411→    eff.includes('productcheckoutwait') ||
   412→   412→    eff.includes('employeeitemplacewait') ||
   413→   413→    eff.includes('selfcheckoutwait')
   414→   414→  )
   415→   415→    tags.push('throughput')
   416→   416→  if (eff.includes('allowedsimultaneoussales')) tags.push('sales-cap')
   417→   417→  if (eff.includes('extracustomersperk') || eff.includes('bystander')) tags.push('customer-volume')
   418→   418→  if (eff.includes('recycle')) tags.push('recycling')
   419→   419→  if (eff.includes('electricfactor') || eff.includes('autopay')) tags.push('finance')
   420→   420→  if (eff.includes('rerollsperday')) tags.push('ordering')
   421→   421→
   422→   422→  return { category, tags }
   423→   423→}

   444→   444→  snapshot: SaveSnapshot | null,
   445→   445→  mode: StrategyMode,
   446→   446→  count: number,
   447→   447→): NextSkillRecommendation[] {
   448→   448→  const unlocked = getUnlockedSet(snapshot)
   449→   449→  const weights = MODE_WEIGHTS[mode]
   450→   450→  const employeeCount = snapshot?.employees?.length ?? 0
   451→   451→  const speedSkillsUnlocked = ENC.skills.filter(
   452→   452→    (s) => s.perk != null && unlocked.has(s.perk) && s.effect.toLowerCase().includes('speed'),
   453→   453→  ).length
   454→   454→  const checkoutSkillsUnlocked = ENC.skills.filter(
   455→   455→    (s) => s.perk != null && unlocked.has(s.perk) && /checkout|selfcheckout|productcheckout/.test(s.effect.toLowerCase()),
   456→   456→  ).length
   457→   457→  const customerSkillsUnlocked = ENC.skills.filter(
   458→   458→    (s) => s.perk != null && unlocked.has(s.perk) && /extracustomersperk|bystander/.test(s.effect.toLowerCase()),
   459→   459→  ).length
   460→   460→  const recycleSkillsUnlocked = ENC.skills.filter(
   461→   461→    (s) => s.perk != null && unlocked.has(s.perk) && s.effect.toLowerCase().includes('recycle'),
   462→   462→  ).length
   463→   463→
   464→   464→  const candidates = ENC.skills.filter((s) => s.perk != null && !unlocked.has(s.perk))
   465→   465→
   466→   466→  const scored = candidates.map((s) => {
   467→   467→    const { tags } = categorizeSkill(s)
   468→   468→    let score = 0.5 // base
   469→   469→    for (const w of weights) {
   470→   470→      if (tags.includes(w.tag)) score += w.weight
   471→   471→    }
   472→   472→    // Slight bonus for skills with concrete numeric effects.
   473→   473→    const parsed = parseEffectForMetric(s.effect)
   474→   474→    if (parsed.length > 0 && parsed[0].delta !== '未提取') score += 0.3
   475→   475→    return { skill: s, score, tags }
   476→   476→  })
   477→   477→  scored.sort((a, b) => b.score - a.score)
   478→   478→
   479→   479→  return scored.slice(0, count).map((x, i) => ({
   480→   480→    skill: x.skill,
   481→   481→    rank: i + 1,
   482→   482→    score: x.score,
   483→   483→    reason: generateReason(x.skill, mode, {
   484→   484→      employeeCount,
   485→   485→      speedSkillsUnlocked,
   486→   486→      checkoutSkillsUnlocked,
   487→   487→      customerSkillsUnlocked,
   488→   488→      recycleSkillsUnlocked,
   489→   489→    }),
   490→   490→  }))
   491→   491→}

   501→   501→function generateReason(
   502→   502→  skill: Skill,
   503→   503→  mode: StrategyMode,
   504→   504→  ctx: SaveContext,
   505→   505→): string {
   506→   506→  const eff = skill.effect
   507→   507→  const parsed = parseEffectForMetric(eff)
   508→   508→  const delta = parsed[0]?.delta ?? ''
   509→   509→  const metric = parsed[0]?.metric ?? ''
   510→   510→  switch (mode) {
   511→   511→    case 'employee':
   512→   512→      if (eff.includes('extraemployeespeedfactor')) {
   513→   513→        return `你目前有 ${ctx.employeeCount} 名員工，此技能讓${metric} ${delta}，直接提升補貨與結帳效率，員工越多加成越大。`
   514→   514→      }
   515→   515→      if (eff.includes('maxemployees')) {
   516→   516→        return `目前員工 ${ctx.employeeCount} 名，此技能直接 +${eff.match(/maxemployees\s*\+=\s*(\d+)/i)?.[1] ?? 'N'} 最大員工數，擴編人力。`
   517→   517→      }
   518→   518→      if (eff.includes('employeeitemplacewait')) {
   519→   519→        return `降低員工補貨等待 ${delta}，補貨更順暢。`
   520→   520→      }
   521→   521→      return `${metric || '效果'} ${delta || eff} — 員工效率相關加成。`
   522→   522→    case 'customer':
   523→   523→      if (eff.includes('extracustomersperk')) {
   524→   524→        return `已解鎖 ${ctx.customerSkillsUnlocked} 個客流技能，再加一個 ${delta} 客流的技能可進一步放大來客數。`
   525→   525→      }
   526→   526→      if (eff.includes('allowedsimultaneoussales')) {
   527→   527→        return `客流增加後結帳壓力上升，此技能 ${delta} 同時結帳數避免塞車。`
   528→   528→      }
   529→   529→      if (eff.includes('bystander')) {
   530→   530→        return `將路人轉為顧客，直接擴大客流池。`
   531→   531→      }
   532→   532→      return `${metric || '效果'} ${delta || eff} — 客流相關加成。`
   533→   533→    case 'checkout':
   534→   534→      if (eff.includes('productcheckoutwait') || eff.includes('selfcheckoutwait')) {
   535→   535→        return `已解鎖 ${ctx.checkoutSkillsUnlocked} 個收銀技能，此技能降低結帳等待 ${delta}，吞吐量再升級。`
   536→   536→      }
   537→   537→      if (eff.includes('allowedsimultaneoussales')) {
   538→   538→        return `增加同時結帳數 ${delta}，分散尖峰壓力。`
   539→   539→      }
   540→   540→      if (eff.includes('extracheckoutmoney')) {
   541→   541→        return `每筆結帳金額 ${delta}，直接放大單客營收。`
   542→   542→      }
   543→   543→      return `${metric || '效果'} ${delta || eff} — 收銀相關加成。`
   544→   544→    case 'recycling':
   545→   545→      if (eff.includes('boxrecyclefactor')) {
   546→   546→        return `已解鎖 ${ctx.recycleSkillsUnlocked} 個回收技能，此技能將回收倍率設為 ${delta}，回收收益倍增。`
   547→   547→      }
   548→   548→      if (eff.includes('recycle')) {
   549→   549→        return `強化回收鏈（${metric} ${delta}），降低廢棄成本。`
   550→   550→      }
   551→   551→      return `${metric || '效果'} ${delta || eff} — 回收相關加成。`
   552→   552→  }
   553→   553→}

   702→   702→/**
   703→   703→ * Return 3-5 tailored skill recommendations based on the store profile.

   705→   705→ */
   706→   706→export function tailoredRecommendations(
   707→   707→  snapshot: SaveSnapshot | null,
   708→   708→  profile: StoreProfile,
   709→   709→  count = 5,
   710→   710→): TailoredRecommendation[] {
   711→   711→  const unlocked = getUnlockedSet(snapshot)
   712→   712→  const archetype = profile.archetype
   713→   713→  // Pick a tag-filter based on the archetype.
   714→   714→  let allowedTags: string[] = []
   715→   715→  if (archetype.includes('廣場') || archetype.includes('大型')) {
   716→   716→    allowedTags = ['speed', 'headcount', 'throughput', 'sales-cap']
   717→   717→  } else if (archetype.includes('小型') || archetype.includes('精簡')) {
   718→   718→    allowedTags = ['speed', 'throughput', 'automation']
   719→   719→  } else if (archetype.includes('中後期')) {
   720→   720→    allowedTags = ['headcount', 'recycling', 'finance']
   721→   721→  } else {
   722→   722→    allowedTags = ['speed', 'throughput', 'sales-cap', 'headcount']
   723→   723→  }
   724→   724→
   725→   725→  const candidates = ENC.skills
   726→   726→    .filter((s) => s.perk != null)
   727→   727→    .map((s) => {
   728→   728→      const { tags } = categorizeSkill(s)
   729→   729→      const matchingTags = tags.filter((t) => allowedTags.includes(t))
   730→   730→      return { skill: s, tags, matchingTags, unlocked: unlocked.has(s.perk as number) }
   731→   731→    })
   732→   732→    .filter((x) => x.matchingTags.length > 0)
   733→   733→
   734→   734→  // Prefer unlocked=false first, then by matchingTags count desc.
   735→   735→  candidates.sort((a, b) => {
   736→   736→    if (a.unlocked !== b.unlocked) return a.unlocked ? 1 : -1
   737→   737→    return b.matchingTags.length - a.matchingTags.length
   738→   738→  })
   739→   739→
   740→   740→  return candidates.slice(0, count).map((x) => ({
   741→   741→    skill: x.skill,
   742→   742→    unlocked: x.unlocked,
   743→   743→    reason: generateTailoredReason(x.skill, x.tags, archetype, profile.metrics),
   744→   744→  }))
   745→   745→}

   822→   822→/**
   823→   823→ * Compute FP "still needed" for a build plan:
   824→   824→ *   (number of plan skills not yet unlocked) × 1000 − current unspent FP
   825→   825→ * (floored at 0 — if the user has enough, returns 0).
   826→   826→ */
   827→   827→export function computeFpNeededForPlan(
   828→   828→  planSkillIds: string[],
   829→   829→  snapshot: SaveSnapshot | null,
   830→   830→): number {
   831→   831→  const unlocked = getUnlockedSet(snapshot)
   832→   832→  let needBuy = 0
   833→   833→  for (const id of planSkillIds) {
   834→   834→    const s = SKILL_BY_ID.get(id)
   835→   835→    if (!s || s.perk == null) continue
   836→   836→    if (!unlocked.has(s.perk)) needBuy++
   837→   837→  }
   838→   838→  const cost = needBuy * FP_COST_PER_SKILL
   839→   839→  const available = snapshot?.franchisePoints ?? 0
   840→   840→  return Math.max(0, cost - available)
   841→   841→}
