   263→// Product selector — search input + filtered dropdown
   264→// ============================================================
   265→function ProductSelector({
   266→  selectedId,
   267→  onSelect,
   268→}: {
   269→  selectedId: number | null
   270→  onSelect: (id: number) => void
   271→}) {
   272→  const [query, setQuery] = useState('')
   273→  const [open, setOpen] = useState(false)
   274→
   275→  const filtered = useMemo(() => {
   276→    const q = query.trim().toLowerCase()
   277→    const list = q
   278→      ? ENC.products.filter(
   279→          (p) =>
   280→            p.name.en.toLowerCase().includes(q) ||
   281→            p.name.zhHant.toLowerCase().includes(q) ||
   282→            p.brand.toLowerCase().includes(q) ||
   283→            String(p.id) === q,
   284→        )
   285→      : ENC.products
   286→    return list.slice(0, 24)
   287→  }, [query])
   288→
   289→  const selected = selectedId != null ? productById.get(selectedId) : null
   290→
   291→  return (
   292→    <Card>
   293→      <CardContent className="p-3">
   294→        <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   295→          <div className="flex items-center gap-2 text-sm font-medium shrink-0">
   296→            <Search className="h-4 w-4 text-muted-foreground" />
   297→            <span>商品搜尋</span>
   298→          </div>
   299→          <div className="relative flex-1">
   300→            <Input
   301→              value={query}
   302→              placeholder="搜尋商品名稱 / brand / ID…"
   303→              onChange={(e) => {
   304→                setQuery(e.target.value)
   305→                setOpen(true)
   306→              }}
   307→              onFocus={() => setOpen(true)}
   308→              onBlur={() => setTimeout(() => setOpen(false), 150)}
   309→              className="h-9"
   310→            />
   311→            {open && filtered.length > 0 && (
   312→              <div className="absolute z-30 mt-1 max-h-72 w-full overflow-y-auto rounded-md border bg-popover shadow-md scrollbar-thin">
   313→                {filtered.map((p) => (
   314→                  <button
   315→                    key={p.id}
   316→                    onMouseDown={(e) => {
   317→                      e.preventDefault()
   318→                      onSelect(p.id)
   319→                      setQuery('')
   320→                      setOpen(false)
   321→                    }}
   322→                    className="flex w-full items-center justify-between gap-2 border-b last:border-b-0 px-3 py-2 text-left text-sm hover:bg-accent"
   323→                  >
   324→                    <div className="min-w-0">
   325→                      <div className="truncate font-medium">{p.name.zhHant} <span className="text-muted-foreground">/ {p.name.en}</span></div>
   326→                      <div className="truncate text-xs text-muted-foreground">#{p.id} · {p.brand} · {p.groupName.zhHant}</div>
   327→                    </div>
   328→                    <span className="shrink-0 font-mono text-xs text-muted-foreground">${fmt(p.basePricePerUnit, 2)}</span>
   329→                  </button>
   330→                ))}
   331→              </div>
   332→            )}
   333→          </div>
   334→          {selected && (
   335→            <Badge variant="outline" className="shrink-0 gap-1">
   336→              <Check className="h-3 w-3 text-emerald-500" />
   337→              {selected.name.zhHant} #{selected.id}
   338→            </Badge>
   339→          )}
   340→        </div>
   341→      </CardContent>
   342→    </Card>
   343→  )
   344→}

   425→// Player price editor
   426→// ============================================================
   427→function PlayerPriceEditor({
   428→  product,
   429→  playerPrice,
   430→  base,
   431→  balanced,
   432→  hasSnapshot,
   433→  onApply,
   434→  onLoadDemo,
   435→}: {
   436→  product: Product | null
   437→  playerPrice: number
   438→  base: number
   439→  balanced: number
   440→  hasSnapshot: boolean
   441→  onApply: (productId: number, price: number) => void
   442→  onLoadDemo: () => void
   443→}) {
   444→  const [draft, setDraft] = useState<string>(String(playerPrice))
   445→
   446→  if (!product) return null
   447→
   448→  const markup = base > 0 ? (playerPrice - base) / base : 0
   449→  const draftNum = Number(draft)
   450→  const draftValid = Number.isFinite(draftNum) && draftNum >= 0
   451→  const draftMarkup = base > 0 && draftValid ? (draftNum - base) / base : 0
   452→
   453→  return (
   454→    <Card>
   455→      <CardHeader className="pb-2">
   456→        <CardTitle className="flex items-center justify-between text-base">
   457→          <span>目前玩家定價</span>
   458→          <ConfidenceBadge
   459→            confidence={hasSnapshot ? 'confirmed' : 'demo'}
   460→            formula="snapshot.productPlayerPricing[productId]"
   461→            note={hasSnapshot ? undefined : '尚未上傳存檔，顯示商品預設 playerPrice'}
   462→          />
   463→        </CardTitle>
   464→      </CardHeader>
   465→      <CardContent>
   466→        <div className="flex flex-col gap-3 sm:flex-row sm:items-end">
   467→          <div className="flex-1">
   468→            <label className="mb-1 block text-xs font-medium text-muted-foreground">
   469→              {product.name.zhHant} #{product.id} — 單價
   470→            </label>
   471→            <Input
   472→              type="number"
   473→              step="0.01"
   474→              min="0"
   475→              value={draft}
   476→              onChange={(e) => setDraft(e.target.value)}
   477→              disabled={!hasSnapshot}
   478→              className="h-9 font-mono"
   479→            />
   480→            {!hasSnapshot && (
   481→              <p className="mt-1 text-xs text-amber-600 dark:text-amber-400">
   482→                未上傳存檔 — 編輯已停用。點擊「載入 Demo」可試用。
   483→              </p>
   484→            )}
   485→          </div>
   486→          <div className="flex gap-2">
   487→            <Button
   488→              size="sm"
   489→              disabled={!hasSnapshot || !draftValid || draftNum === playerPrice}
   490→              onClick={() => onApply(product.id, draftNum)}
   491→            >
   492→              <Check className="mr-1 h-3.5 w-3.5" /> 套用
   493→            </Button>
   494→            <Button
   495→              size="sm"
   496→              variant="outline"
   497→              disabled={!hasSnapshot}
   498→              onClick={() => onApply(product.id, balanced)}
   499→            >
   500→              <Wand2 className="mr-1 h-3.5 w-3.5" /> 套用 Balanced (${fmt(balanced, 2)})
   501→            </Button>
   502→            {!hasSnapshot && (
   503→              <Button size="sm" variant="secondary" onClick={onLoadDemo}>
   504→                載入 Demo
   505→              </Button>
   506→            )}
   507→          </div>
   508→        </div>
   509→        <div className="mt-3 grid grid-cols-2 gap-2 sm:grid-cols-4">
   510→          <StatCard label="Base" value={fmtMoney(base)} confidence="confirmed" accent="neutral" />
   511→          <StatCard
   512→            label="目前 Player"
   513→            value={fmtMoney(playerPrice)}
   514→            confidence={hasSnapshot ? 'confirmed' : 'demo'}
   515→            accent="neutral"
   516→          />
   517→          <StatCard
   518→            label="Markup %"
   519→            value={fmt(markup * 100, 1)}
   520→            unit="%"
   521→            confidence={hasSnapshot ? 'confirmed' : 'demo'}
   522→            formula="(playerPrice − base) / base"
   523→            accent={markup > 0.3 ? 'bad' : markup > 0.05 ? 'warn' : 'good'}
   524→          />
   525→          {draftValid && draftNum !== playerPrice && (
   526→            <StatCard
   527→              label="草稿 Markup"
   528→              value={fmt(draftMarkup * 100, 1)}
   529→              unit="%"
   530→              confidence="unverified"
   531→              formula="(draft − base) / base"
   532→              accent={draftMarkup > 0.3 ? 'bad' : draftMarkup > 0.05 ? 'warn' : 'good'}
   533→            />
   534→          )}
   535→        </div>
   536→      </CardContent>
   537→    </Card>
   538→  )
   539→}

   542→// Room vote panel
   543→// ============================================================
   544→function RoomVotePanel({
   545→  product,
   546→  balancedPrice,
   547→  oldPrice,
   548→  members,
   549→  selfId,
   550→  onProposeExperiment,
   551→}: {
   552→  product: Product
   553→  balancedPrice: number
   554→  oldPrice: number
   555→  members: { id: string; name: string; color: string; role: string }[]
   556→  selfId: string
   557→  onProposeExperiment: (e: PriceExperiment) => void
   558→}) {
   559→  // votes: memberId -> 'approve' | 'reject' | null (local-only tally)
   560→  const [votes, setVotes] = useState<Record<string, 'approve' | 'reject'>>({})
   561→
   562→  const approveCount = Object.values(votes).filter((v) => v === 'approve').length
   563→  const rejectCount = Object.values(votes).filter((v) => v === 'reject').length
   564→  const totalVotes = approveCount + rejectCount
   565→
   566→  const setVote = (mid: string, v: 'approve' | 'reject') => {

   577→      </CardHeader>
   578→      <CardContent className="space-y-3">
   579→        <div className="rounded-md border bg-muted/30 p-3 text-sm">
   580→          <div className="flex flex-wrap items-center justify-between gap-2">
   581→            <div>
   582→              <span className="font-semibold">{product.name.zhHant}</span>{' '}
   583→              <span className="text-muted-foreground">#{product.id}</span>

   589→            </div>
   590→          </div>
   591→        </div>
   592→
   593→        <div className="flex flex-wrap gap-2">
   594→          {members.map((m) => {
   595→            const v = votes[m.id]
   596→            return (
   597→              <div
   598→                key={m.id}
   599→                className="flex items-center gap-2 rounded-md border bg-card px-2 py-1.5"
   600→                style={{ borderLeft: `3px solid ${m.color}` }}
   601→              >
   602→                <span className="text-xs font-medium">{m.name}{m.id === selfId ? ' (你)' : ''}</span>

   604→                  size="sm"
   605→                  variant={v === 'approve' ? 'default' : 'outline'}
   606→                  className="h-7 px-2 text-xs"
   607→                  onClick={() => setVote(m.id, 'approve')}
   608→                >
   609→                  <ThumbsUp className="mr-1 h-3 w-3" /> 贊成
   610→                </Button>
   611→                <Button
   612→                  size="sm"
   613→                  variant={v === 'reject' ? 'default' : 'outline'}
   614→                  className="h-7 px-2 text-xs"
   615→                  onClick={() => setVote(m.id, 'reject')}
   616→                >
   617→                  <ThumbsDown className="mr-1 h-3 w-3" /> 反對
   618→                </Button>
   619→              </div>
   620→            )
   621→          })}
   622→        </div>
   623→
   624→        <div className="flex items-center gap-3 rounded-md border p-2 text-sm">
   625→          <Badge variant="outline" className="border-emerald-500/30 text-emerald-700 dark:text-emerald-300">

   627→          </Badge>
   628→          <Badge variant="outline" className="border-rose-500/30 text-rose-700 dark:text-rose-300">
   629→            <ThumbsDown className="mr-1 h-3 w-3" /> {rejectCount} 反對
   630→          </Badge>
   631→          <span className="text-xs text-muted-foreground">{totalVotes}/{members.length} 已投票</span>
   632→          {approveCount > rejectCount && totalVotes > 0 && (
   633→            <Button
   634→              size="sm"
   635→              variant="default"
   636→              className="ml-auto"
   637→              onClick={() =>
   638→                onProposeExperiment({
   639→                  id: `exp-${product.id}-${Date.now()}`,
   640→                  productId: product.id,
   641→                  oldPrice,
   642→                  newPrice: balancedPrice,
   643→                  observedSales: '',
   644→                  observedComplaints: '',
   645→                  conclusion: '房間投票通過 — 待實測',
   646→                  updatedAt: Date.now(),
   647→                  updatedBy: selfId,
   648→                })
   649→              }
   650→            >
   651→              <Plus className="mr-1 h-3.5 w-3.5" /> 加入實驗追蹤器
   652→            </Button>
   653→          )}
   654→        </div>
   655→      </CardContent>
   656→    </Card>
   657→  )
   658→}

   660→// ============================================================
   661→// Bulk pricing view — top 30 by boxValue
   662→// ============================================================
   663→function BulkPricingView({
   664→  snapshotPricing,
   665→  hasSnapshot,
   666→  onApplyBalanced,
   667→}: {
   668→  snapshotPricing: Record<number, number>
   669→  hasSnapshot: boolean
   670→  onApplyBalanced: (productId: number, balanced: number) => void
   671→}) {
   672→  const rows = useMemo(() => {
   673→    return ENC.products
   674→      .map((p) => {
   675→        const box = computeBoxValue(p).value
   676→        const playerPrice = (snapshotPricing[p.id] as number | undefined) ?? p.playerPrice
   677→        const markup = p.basePricePerUnit > 0 ? (playerPrice - p.basePricePerUnit) / p.basePricePerUnit : 0
   678→        const balanced = p.basePricePerUnit * 1.15
   679→        return { p, box, playerPrice, markup, balanced }
   680→      })
   681→      .sort((a, b) => b.box - a.box)
   682→      .slice(0, 30)
   683→  }, [snapshotPricing])
   684→
   685→  return (
   686→    <Card>
   687→      <CardHeader className="pb-2">
   688→        <CardTitle className="flex items-center justify-between text-base">
   689→          <span>批量定價檢視 — Top 30 by Box Value</span>
   690→          <ConfidenceBadge
   691→            confidence={hasSnapshot ? 'confirmed' : 'demo'}
   692→            formula="boxValue = basePrice × maxItemsPerBox"
   693→            note="balanced = base × 1.15 (needs-runtime)"
   694→          />
   695→        </CardTitle>
   696→      </CardHeader>
   697→      <CardContent>
   698→        <div className="max-h-96 overflow-y-auto scrollbar-thin">
   699→          <Table>
   700→            <TableHeader className="sticky top-0 bg-card z-10">
   701→              <TableRow>
   702→                <TableHead className="min-w-[180px]">商品</TableHead>
   703→                <TableHead className="text-right">Base</TableHead>
   704→                <TableHead className="text-right">Box Value</TableHead>
   705→                <TableHead className="text-right">Player Price</TableHead>
   706→                <TableHead className="text-right">Markup %</TableHead>
   707→                <TableHead className="text-right">Balanced</TableHead>
   708→                <TableHead className="text-right">動作</TableHead>
   709→              </TableRow>
   710→            </TableHeader>
   711→            <TableBody>
   712→              {rows.map((r) => (
   713→                <TableRow key={r.p.id}>
   714→                  <TableCell>
   715→                    <div className="flex flex-col">
   716→                      <span className="truncate font-medium">{r.p.name.zhHant}</span>
   717→                      <span className="text-xs text-muted-foreground">#{r.p.id} · {r.p.brand}</span>
   718→                    </div>
   719→                  </TableCell>
   720→                  <TableCell className="text-right font-mono">${fmt(r.p.basePricePerUnit, 2)}</TableCell>
   721→                  <TableCell className="text-right font-mono">${fmt(r.box, 0)}</TableCell>
   722→                  <TableCell className="text-right font-mono">
   723→                    {hasSnapshot ? `$${fmt(r.playerPrice, 2)}` : <span className="text-muted-foreground">—</span>}
   724→                  </TableCell>
   725→                  <TableCell className={`text-right font-mono ${r.markup > 0.3 ? 'text-rose-600' : r.markup > 0.05 ? 'text-amber-600' : 'text-emerald-600'}`}>
   726→                    {hasSnapshot ? `${fmt(r.markup * 100, 1)}%` : '—'}
   727→                  </TableCell>
   728→                  <TableCell className="text-right font-mono font-semibold">${fmt(r.balanced, 2)}</TableCell>
   729→                  <TableCell className="text-right">
   730→                    <Button
   731→                      size="sm"
   732→                      variant="outline"
   733→                      className="h-7 px-2 text-xs"
   734→                      disabled={!hasSnapshot}
   735→                      onClick={() => onApplyBalanced(r.p.id, r.balanced)}
   736→                    >
   737→                      <Wand2 className="mr-1 h-3 w-3" /> 套用 balanced
   738→                    </Button>
   739→                  </TableCell>
   740→                </TableRow>
   741→              ))}
   742→            </TableBody>
   743→          </Table>
   744→        </div>
   745→      </CardContent>
   746→    </Card>
   747→  )
   748→}

   750→// ============================================================
   751→// Experiment tracker
   752→// ============================================================
   753→function ExperimentTracker({
   754→  roomPricePlan,
   755→  selfName,
   756→  selfId,
   757→  selectedProduct,
   758→  balancedPrice,
   759→  oldPrice,
   760→  onUpsert,
   761→}: {
   762→  roomPricePlan: PriceExperiment[] | null
   763→  selfName: string
   764→  selfId: string
   765→  selectedProduct: Product | null
   766→  balancedPrice: number
   767→  oldPrice: number
   768→  onUpsert: (e: PriceExperiment) => void
   769→}) {
   770→  // Local-only experiments (used when no room).
   771→  const [localExps, setLocalExps] = useState<PriceExperiment[]>([])
   772→  const experiments = roomPricePlan ?? localExps
   773→
   774→  const addExperiment = () => {
   775→    if (!selectedProduct) return
   776→    const exp: PriceExperiment = {
   777→      id: `exp-${selectedProduct.id}-${Date.now()}`,
   778→      productId: selectedProduct.id,
   779→      oldPrice,
   780→      newPrice: balancedPrice,
   781→      observedSales: '',
   782→      observedComplaints: '',
   783→      conclusion: '',
   784→      updatedAt: Date.now(),
   785→      updatedBy: selfName || selfId,
   786→    }
   787→    if (roomPricePlan) {
   788→      onUpsert(exp)
   789→    } else {
   790→      setLocalExps((s) => [...s.filter((x) => x.id !== exp.id), exp])
   791→    }
   792→  }
   793→
   794→  const updateField = (id: string, patch: Partial<PriceExperiment>) => {
   795→    const next = experiments.map((e) => (e.id === id ? { ...e, ...patch, updatedAt: Date.now() } : e))
   796→    const target = next.find((e) => e.id === id)
   797→    if (!target) return
   798→    if (roomPricePlan) {
   799→      onUpsert(target)
   800→    } else {
   801→      setLocalExps(next)
   802→    }
   803→  }
   804→
   805→  return (
   806→    <Card>
   807→      <CardHeader className="pb-2">
   808→        <div className="flex flex-wrap items-center justify-between gap-2">
   809→          <CardTitle className="flex items-center gap-2 text-base">
   810→            <FlaskConical className="h-4 w-4 text-fuchsia-500" />
   811→            價格實驗追蹤器
   812→          </CardTitle>
   813→          <div className="flex items-center gap-2">
   814→            <ConfidenceBadge
   815→              confidence="unverified"
   816→              note="手動玩家觀察結果，非自動擷取"
   817→            />
   818→            <Badge variant="outline" className="text-xs">
   819→              {roomPricePlan ? `房間模式 · ${experiments.length}` : `本地模式 · ${experiments.length}`}
   820→            </Badge>
   821→            <Button size="sm" onClick={addExperiment} disabled={!selectedProduct}>
   822→              <Plus className="mr-1 h-3.5 w-3.5" /> 新增實驗
   823→            </Button>
   824→          </div>
   825→        </div>
   826→      </CardHeader>
   827→      <CardContent>
   828→        {experiments.length === 0 ? (
   829→          <div className="py-6 text-center text-sm text-muted-foreground">
   830→            尚無實驗。選擇商品後點「新增實驗」開始追蹤你的價格嘗試與觀察結果。
   831→          </div>
   832→        ) : (
   833→          <div className="max-h-96 overflow-y-auto scrollbar-thin">
   834→            <Table>
   835→              <TableHeader className="sticky top-0 bg-card z-10">
   836→                <TableRow>
   837→                  <TableHead className="min-w-[160px]">商品</TableHead>
   838→                  <TableHead className="text-right">舊價</TableHead>
   839→                  <TableHead className="text-right">新價</TableHead>
   840→                  <TableHead className="min-w-[120px]">觀察銷量</TableHead>
   841→                  <TableHead className="min-w-[120px]">客訴</TableHead>
   842→                  <TableHead className="min-w-[160px]">結論</TableHead>
   843→                  <TableHead className="text-right">更新時間</TableHead>
   844→                  <TableHead>更新者</TableHead>
   845→                </TableRow>
   846→              </TableHeader>
   847→              <TableBody>
   848→                {experiments.map((e) => {
   849→                  const p = productById.get(e.productId)
   850→                  const delta = e.oldPrice > 0 ? ((e.newPrice - e.oldPrice) / e.oldPrice) * 100 : 0
   851→                  return (
   852→                    <TableRow key={e.id}>
   853→                      <TableCell>
   854→                        <div className="flex flex-col">
   855→                          <span className="truncate font-medium">{p?.name.zhHant ?? `#${e.productId}`}</span>
   856→                          <span className="text-xs text-muted-foreground">#{e.productId}</span>
   857→                        </div>
   858→                      </TableCell>
   859→                      <TableCell className="text-right font-mono">${fmt(e.oldPrice, 2)}</TableCell>
   860→                      <TableCell className="text-right font-mono">
   861→                        ${fmt(e.newPrice, 2)}
   862→                        <div className={`text-[10px] ${delta > 0 ? 'text-amber-600' : delta < 0 ? 'text-emerald-600' : 'text-muted-foreground'}`}>
   863→                          {delta > 0 ? '+' : ''}{fmt(delta, 1)}%
   864→                        </div>
   865→                      </TableCell>
   866→                      <TableCell>
   867→                        <Input
   868→                          value={e.observedSales}
   869→                          placeholder="例: 12 units/day"
   870→                          onChange={(ev) => updateField(e.id, { observedSales: ev.target.value })}
   871→                          className="h-8 text-xs"
   872→                        />
   873→                      </TableCell>
   874→                      <TableCell>
   875→                        <Input
   876→                          value={e.observedComplaints}
   877→                          placeholder="例: 3 客訴"
   878→                          onChange={(ev) => updateField(e.id, { observedComplaints: ev.target.value })}
   879→                          className="h-8 text-xs"
   880→                        />
   881→                      </TableCell>
   882→                      <TableCell>
   883→                        <Input
   884→                          value={e.conclusion}
   885→                          placeholder="例: 銷量持平，客訴上升 — 退回原價"
   886→                          onChange={(ev) => updateField(e.id, { conclusion: ev.target.value })}
   887→                          className="h-8 text-xs"
   888→                        />
   889→                      </TableCell>
   890→                      <TableCell className="text-right text-xs text-muted-foreground whitespace-nowrap">
   891→                        {new Date(e.updatedAt).toLocaleString(undefined, { hour12: false })}
   892→                      </TableCell>
   893→                      <TableCell className="text-xs whitespace-nowrap">{e.updatedBy}</TableCell>
   894→                    </TableRow>
   895→                  )
   896→                })}
   897→              </TableBody>
   898→            </Table>
   899→          </div>
   900→        )}
   901→        <div className="mt-2 flex items-start gap-2 rounded-md border bg-muted/30 p-2 text-xs text-muted-foreground">
   902→          <Info className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   903→          <span>
   904→            實驗結果為手動輸入的玩家觀察。建議流程：套用新價 → 在遊戲內觀察 1 天 → 記錄銷量/客訴 → 寫下結論。
   905→            若有房間，實驗會同步到 <code className="font-mono">room.pricePlan</code>，供隊友查看。
   906→          </span>
   907→        </div>
   908→      </CardContent>
   909→    </Card>
   910→  )
   911→}
