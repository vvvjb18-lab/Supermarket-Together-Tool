   110→// ---------- Helpers ----------
   111→function percentile(sorted: number[], p: number): number {
   112→  if (sorted.length === 0) return 0
   113→  const idx = (sorted.length - 1) * p
   114→  const lo = Math.floor(idx)
   115→  const hi = Math.ceil(idx)
   116→  if (lo === hi) return sorted[lo]
   117→  return sorted[lo] + (sorted[hi] - sorted[lo]) * (idx - lo)
   118→}

   160→      const weighted = computeWeightedBoxProxy(p, demand).value

   166→      for (const n of ENC.necessities) {
   167→        if (n.rawTokens.includes(p.id)) necessities.push(n.index)
   168→      }
   169→      const mfgIdx = ENC.manufacturingProducts.findIndex((m) => m.linkedProductID === p.id)
   170→      return {
   171→        p,
   172→        boxValue,
   173→        colliderVolume,
   174→        valueDensity,
   175→        boxValueDensity,
   176→        demandProxy: demand,
   177→        weightedValue: weighted,
   178→        role: 'unknown' as ProductRole,
   179→        isPremium: ENC.premiumProducts.includes(p.id),
   180→        isStackable: p.isStackable,
   181→        seasons,
   182→        necessities,
   183→        manufacturingLink: mfgIdx >= 0 ? mfgIdx : null,
   184→        inventory: (inv[p.id] as number) ?? 0,
   185→        shelfProps: propMap.get(p.id) ?? [],
   186→      }
   187→    })
   188→
   189→    // percentiles for role classification
   190→    const boxValues = raw.map((r) => r.boxValue).sort((a, b) => a - b)

   192→    const demands = raw.map((r) => r.demandProxy).sort((a, b) => a - b)
   193→    const p = {
   194→      boxP75: percentile(boxValues, 0.75),
   195→      densP75: percentile(densities, 0.75),
   196→      densP25: percentile(densities, 0.25),
   197→      demP25: percentile(demands, 0.25),
   198→      demMedian: percentile(demands, 0.5),
   199→    }
   200→
   201→    for (const r of raw) {
   202→      let role: ProductRole = 'unknown'
   203→      if (r.isPremium) role = 'premium'
   204→      else if (r.seasons.length > 0) role = 'seasonal'
   205→      else if (r.boxValue > p.boxP75) role = 'high-value'
   206→      else if (r.valueDensity > p.densP75) role = 'high-density'
   207→      else if (r.p.tier <= 5 && r.demandProxy > p.demMedian) role = 'staple'
   208→      else if (r.valueDensity < p.densP25 && r.demandProxy < p.demP25) role = 'trap'
   209→      r.role = role
   210→    }
   211→
   212→    const stats = {
   213→      total: raw.length,
   214→      premium: raw.filter((r) => r.isPremium).length,
   215→      seasonal: raw.filter((r) => r.seasons.length > 0).length,
   216→      stackable: raw.filter((r) => r.isStackable).length,
   217→      avgBoxValue: raw.reduce((s, r) => s + r.boxValue, 0) / raw.length,
   218→    }

   289→        case 'weightedValue': return r.weightedValue
   290→        case 'boxValueDensity': return r.boxValueDensity
   291→      }

   295→      const bv = sortVal(b)

   311→      }

   329→    setMissingHighDemand(false)

   331→    setHighDensity(false)
   332→    setLowValueTrap(false)

   336→    text !== '' ||
   337→    group !== 'all' ||
   338→    tier !== 'all' ||
   339→    season !== 'all' ||
   340→    necessity !== 'all' ||
   341→    premiumOnly ||
   342→    stackableOnly ||
   343→    unlockedOnly ||
   344→    stockedOnly ||
   345→    missingHighDemand ||
   346→    highBoxValue ||
   347→    highDensity ||
   348→    lowValueTrap
   349→
   350→  return (
   351→    <div className="mx-auto max-w-[1700px] space-y-4 p-4">
   352→      <div>
   353→        <h1 className="text-2xl font-bold tracking-tight">商品百科</h1>
   354→        <p className="text-sm text-muted-foreground">
   355→          全 {stats.total} 項商品 · 來源：encyclopedia.json (products.tsv + NPC_Manager reflection)
   356→        </p>
   357→      </div>
   358→
   359→      {/* Stat row */}
   360→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-5">
   361→        <StatCard label="總商品數" value={stats.total} confidence="confirmed" formula="encyclopedia.products.length" />
   362→        <StatCard label="Premium 商品" value={stats.premium} confidence="confirmed" formula="premiumProducts.length" accent="good" hint="encyclopedia.premiumProducts" />
   363→        <StatCard label="季節性商品" value={stats.seasonal} confidence="confirmed" formula="Σ seasons[i].productIds ∩ products" accent="neutral" />
   364→        <StatCard label="可堆疊" value={stats.stackable} confidence="confirmed" formula="products.isStackable=true" accent="neutral" />
   365→        <StatCard
   366→          label="平均單箱價值"
   367→          value={fmtMoney(stats.avgBoxValue)}
   368→          confidence="confirmed"
   369→          formula="avg(boxValue) where boxValue=basePrice×maxItemsPerBox"
   370→          accent="good"
   371→        />
   372→      </div>
   373→
   374→      {/* Filter bar */}
   375→      <Card>
   376→        <CardHeader className="pb-3">
   377→          <CardTitle className="flex items-center justify-between text-base">
   378→            <span className="flex items-center gap-2">
   379→              <Search className="h-4 w-4" /> 篩選與排序
   380→            </span>
   381→            <div className="flex items-center gap-2">
   382→              <Badge variant="outline" className="text-[10px]">顯示 {filtered.length} / {rows.length}</Badge>
   383→              {anyFilterActive && (
   384→                <Button size="sm" variant="ghost" onClick={resetFilters} className="h-7 px-2 text-xs">
   385→                  <X className="mr-1 h-3 w-3" /> 清除
   386→                </Button>
   387→              )}
   388→            </div>
   389→          </CardTitle>
   390→        </CardHeader>
   391→        <CardContent className="space-y-3">
   392→          <div className="flex flex-wrap items-center gap-2">
   393→            <Input
   394→              value={text}
   395→              onChange={(e) => setText(e.target.value)}
   396→              placeholder="搜尋 ID / 名稱 / 品牌…"
   397→              className="h-8 w-56 text-sm"
   398→            />
   399→            <Select value={group} onValueChange={setGroup}>
   400→              <SelectTrigger size="sm" className="h-8 w-44 text-xs"><SelectValue placeholder="商品群組" /></SelectTrigger>

   444→            <ToggleChip label="低價值陷阱 (P25)" checked={lowValueTrap} onChange={setLowValueTrap} />

   449→      {/* Table */}

   486→                    <TableCell className="text-xs text-muted-foreground">{r.p.brand}</TableCell>
   487→                    <TableCell className="text-xs">

   494→                              : '#888',

   496→                        />

   498→                      </span>
   499→                    </TableCell>

   501→                    <TableCell className="text-right font-mono text-xs">{fmtMoney(r.p.basePricePerUnit)}</TableCell>
   502→                    <TableCell className="text-right font-mono text-xs">{r.p.maxItemsPerBox}</TableCell>
   503→                    <TableCell className="text-right font-mono text-xs">{fmtMoney(r.boxValue)}</TableCell>
   504→                    <TableCell className="text-right font-mono text-xs">{fmt(r.colliderVolume, 5)}</TableCell>
   505→                    <TableCell className="text-right font-mono text-xs">{fmt(r.valueDensity, 2)}</TableCell>
   506→                    <TableCell className="text-right font-mono text-xs">{fmt(r.demandProxy, 5)}</TableCell>
   507→                    <TableCell className="text-right font-mono text-xs">{fmt(r.weightedValue, 3)}</TableCell>
   508→                    <TableCell className="text-right font-mono text-xs">{fmt(r.boxValueDensity, 2)}</TableCell>
   509→                    <TableCell>
   510→                      <Badge variant="outline" className={`text-[10px] ${ROLE_STYLE[r.role]}`}>
   511→                        {ROLE_LABEL[r.role]}
   512→                      </Badge>
   513→                    </TableCell>
   514→                  </TableRow>
   515→                ))}
   516→                {filtered.length === 0 && (
   517→                  <TableRow>
   518→                    <TableCell colSpan={14} className="py-8 text-center text-sm text-muted-foreground">
   519→                      無符合條件的商品。調整篩選器後重試。
   520→                    </TableCell>
   521→                  </TableRow>
   522→                )}
   523→              </TableBody>
   524→            </Table>
   525→          </div>
   526→        </CardContent>
   527→      </Card>
   528→
   529→      <Sheet
   530→        open={openId !== null}
   531→        onOpenChange={(o) => {
   532→          if (!o) setSelectedProduct(null)
   533→        }}
   534→      >
   535→        <SheetContent side="right" className="w-full overflow-y-auto sm:max-w-lg">
   536→          {openRow && <ProductDetail row={openRow} pct={pct} onGoToProfit={() => setView('profit')} />}
   537→        </SheetContent>
   538→      </Sheet>
   539→    </div>
   540→  )
   541→}

   610→  const proxyConf: Confidence = 'proxy'

   613→  const roleFormula =

   619→    : row.role === 'trap' ? `valueDensity ${fmt(row.valueDensity)} < P25 ${fmt(pct.densP25)} ∧ demandProxy ${fmt(row.demandProxy, 5)} < P25 ${fmt(pct.demP25, 5)}`
   620→    : 'falls through all role rules'
