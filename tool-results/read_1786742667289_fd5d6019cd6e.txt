   168→function curvedPath(a: AtlasNode, b: AtlasNode): string {
   169→  const dx = b.x - a.x
   170→  const dy = b.y - a.y
   171→  const dist = Math.sqrt(dx * dx + dy * dy) || 1
   172→  // perpendicular offset for the control point — gives a gentle arc
   173→  const offset = Math.min(60, dist * 0.18)
   174→  const mx = (a.x + b.x) / 2
   175→  const my = (a.y + b.y) / 2
   176→  const cx = mx + (-dy / dist) * offset
   177→  const cy = my + (dx / dist) * offset
   178→  return `M ${a.x} ${a.y} Q ${cx} ${cy} ${b.x} ${b.y}`
   179→}
   180→
   181→function edgeThickness(count: number): number {
   182→  // log-scale thickness between 0.6 and 5
   183→  const t = 0.6 + Math.log2(count + 1) * 0.5
   184→  return Math.min(t, 5)
   185→}

   188→  const [selected, setSelected] = useState<EntityTypeKey | null>(null)
   189→  const [hovered, setHovered] = useState<EntityTypeKey | null>(null)
   190→  const [hoveredEdge, setHoveredEdge] = useState<number | null>(null)

   193→  const [drag, setDrag] = useState<{ x: number; y: number; px: number; py: number } | null>(null)
   194→
   195→  const active = hovered ?? selected

   197→    if (!active) return new Set<number>()
   198→    const set = new Set<number>()
   199→    ATLAS_EDGES.forEach((e, i) => {
   200→      if (e.from === active || e.to === active) set.add(i)
   201→    })
   202→    return set
   203→  }, [active])
   204→
   205→  const relatedNodes = useMemo(() => {
   206→    if (!active) return new Set<EntityTypeKey>()
   207→    const set = new Set<EntityTypeKey>([active])
   208→    ATLAS_EDGES.forEach((e) => {
   209→      if (e.from === active) set.add(e.to)
   210→      if (e.to === active) set.add(e.from)
   211→    })
   212→    return set
   213→  }, [active])
   214→
   215→  const onPointerDown = useCallback(
   216→    (e: ReactPointerEvent<SVGSVGElement>) => {
   217→      if (e.button !== 0) return
   218→      const target = e.target as SVGElement
   219→      // only pan when clicking on empty SVG (not on a node/edge)
   220→      if (target.tagName === 'circle' || target.tagName === 'path' || target.tagName === 'text') return
   221→      setDrag({ x: e.clientX, y: e.clientY, px: pan.x, py: pan.y })
   222→      ;(e.currentTarget as SVGSVGElement).setPointerCapture(e.pointerId)
   223→    },
   224→    [pan],
   225→  )
   226→  const onPointerMove = useCallback(
   227→    (e: ReactPointerEvent<SVGSVGElement>) => {
   228→      if (!drag) return
   229→      setPan({ x: drag.px + (e.clientX - drag.x), y: drag.py + (e.clientY - drag.y) })
   230→    },
   231→    [drag],
   232→  )
   233→  const onPointerUp = useCallback((e: ReactPointerEvent<SVGSVGElement>) => {
   234→    setDrag(null)
   235→    ;(e.currentTarget as SVGSVGElement).releasePointerCapture(e.pointerId)
   236→  }, [])
   237→
   238→  const onWheel = useCallback((e: React.WheelEvent<SVGSVGElement>) => {
   239→    const delta = e.deltaY > 0 ? 0.9 : 1.1
   240→    setZoom((z) => Math.max(0.4, Math.min(3, z * delta)))
   241→  }, [])
   242→
   243→  const zoomBy = (factor: number) => setZoom((z) => Math.max(0.4, Math.min(3, z * factor)))
   244→  const resetView = () => {
   245→    setZoom(1)
   246→    setPan({ x: 0, y: 0 })
   247→  }
   248→
   249→  const selectedNode = selected ? atlasNodeById(selected) : null
   250→  const selectedInfo = selected ? ENTITY_TYPES.find((e) => e.key === selected) : null
   251→  const selectedRels = selected ? getRelationships(selected) : []
   252→  const selectedSamples = selected ? getEntityTypeSamples(selected, 5) : []
   253→
   254→  return (
   255→    <div className="grid gap-4 lg:grid-cols-[1fr_320px]">
   256→      <Card className="overflow-hidden p-0">
   257→        <div className="flex items-center justify-between gap-2 border-b px-3 py-2">
   258→          <div className="flex items-center gap-2">
   259→            <NetworkIcon className="h-4 w-4 text-fuchsia-500" />
   260→            <span className="text-sm font-semibold">{pick(STR.tab1, lang)}</span>
   261→            <Badge variant="secondary" className="text-[10px]">
   262→              {ATLAS_NODES.length} nodes · {ATLAS_EDGES.length} edges
   263→            </Badge>
   264→          </div>
   265→          <div className="flex items-center gap-1">
   266→            <Button size="sm" variant="ghost" onClick={() => zoomBy(1.2)} title={pick(STR.zoomIn, lang)}>
   267→              <ZoomIn className="h-4 w-4" />
   268→            </Button>
   269→            <Button size="sm" variant="ghost" onClick={() => zoomBy(1 / 1.2)} title={pick(STR.zoomOut, lang)}>
   270→              <ZoomOut className="h-4 w-4" />
   271→            </Button>
   272→            <Button size="sm" variant="ghost" onClick={resetView} title={pick(STR.zoomReset, lang)}>
   273→              <Maximize className="h-4 w-4" />
   274→            </Button>
   275→            {selected && (
   276→              <Button size="sm" variant="ghost" onClick={() => setSelected(null)} title={pick(STR.reset, lang)}>
   277→                <RotateCcw className="h-4 w-4" />
   278→              </Button>
   279→            )}
   280→          </div>
   281→        </div>
   282→        <div className="relative">
   283→          <svg
   284→            viewBox="0 0 1200 800"
   285→            width="100%"
   286→            className="block h-[520px] cursor-grab touch-none select-none active:cursor-grabbing md:h-[640px]"
   287→            style={{ background: 'radial-gradient(ellipse at center, hsl(var(--card)) 0%, hsl(var(--background)) 70%)' }}
   288→            onPointerDown={onPointerDown}
   289→            onPointerMove={onPointerMove}
   290→            onPointerUp={onPointerUp}
   291→            onWheel={onWheel}
   292→          >
   293→            <defs>
   294→              {/* radial gradients per category — lighter center → saturated edge */}
   295→              {ENTITY_CATEGORIES.map((c) => (
   296→                <radialGradient key={c.key} id={`grad-${c.key}`} cx="35%" cy="35%" r="75%">
   297→                  <stop offset="0%" stopColor={c.color} stopOpacity={1} />
   298→                  <stop offset="60%" stopColor={c.color} stopOpacity={0.85} />
   299→                  <stop offset="100%" stopColor={c.color} stopOpacity={0.5} />
   300→                </radialGradient>

   309→
   310→            <g transform={`translate(${pan.x} ${pan.y}) scale(${zoom})`}>
   311→              {/* edges */}
   312→              {ATLAS_EDGES.map((edge, i) => {

   328→                      strokeDasharray={isRelated ? undefined : '4 4'}
   329→                      className="text-muted-foreground transition-all duration-200"
   330→                      style={{ cursor: 'pointer' }}
   331→                      onMouseEnter={() => setHoveredEdge(i)}
   332→                      onMouseLeave={() => setHoveredEdge(null)}

   357→                    if (!from || !to) return null

   388→                    onMouseEnter={() => setHovered(node.id)}
   389→                    onMouseLeave={() => setHovered(null)}
   390→                  >
   391→                    {/* selection ring */}
   392→                    {isSelected && (
   393→                      <circle r={node.r + 6} fill="none" stroke="hsl(var(--primary))" strokeWidth={2.5} strokeOpacity={0.9} />
   394→                    )}
   395→                    {/* hover ring */}
   396→                    {isHovered && !isSelected && (
   397→                      <circle r={node.r + 4} fill="none" stroke={node.color} strokeWidth={1.5} strokeOpacity={0.6} />
   398→                    )}
   399→                    {/* main circle */}
   400→                    <circle
   401→                      r={node.r}
   402→                      fill={`url(#grad-${node.category})`}
   403→                      stroke={node.color}
   404→                      strokeWidth={2}
   405→                      filter="url(#node-shadow)"
   406→                    />
   407→                    {/* inner highlight */}
   408→                    <circle r={node.r * 0.55} cx={-node.r * 0.22} cy={-node.r * 0.22} fill="white" fillOpacity={isHovered ? 0.32 : 0.18} />
   409→                    {/* label */}
   410→                    <text
   411→                      y={node.r + 16}
   412→                      textAnchor="middle"
   413→                      className="pointer-events-none fill-foreground text-[12px] font-semibold"
   414→                    >
   415→                      {node.label}
   416→                    </text>
   417→                    <text
   418→                      y={node.r + 30}
   419→                      textAnchor="middle"
   420→                      className="pointer-events-none fill-muted-foreground text-[10px]"
   421→                    >
   422→                      {(() => {

   433→          {/* legend overlay */}

   436→              {pick(STR.legend, lang)}
   437→            </div>
   438→            <div className="grid grid-cols-2 gap-x-3 gap-y-1">
   439→              {ENTITY_CATEGORIES.map((c) => (
   440→                <div key={c.key} className="flex items-center gap-1.5">
   441→                  <span className="inline-block h-2.5 w-2.5 rounded-full" style={{ background: c.color }} />
   442→                  <span className="text-[10px]">{pick({ zhHant: c.labelZh, en: c.labelEn }, lang)}</span>
   443→                </div>
   444→              ))}
   445→            </div>
   446→          </div>
   447→        </div>
   448→      </Card>
   449→
   450→      {/* side panel */}
   451→      <Card className="h-fit p-4">
   452→        {!selectedNode || !selectedInfo ? (

   563→function ProductHierarchyTree({ lang }: { lang: Lang }) {
   564→  const tree = useMemo<HierarchyGroup[]>(() => productHierarchy(), [])
   565→  const [expanded, setExpanded] = useState<Set<string>>(new Set())
   566→  const [query, setQuery] = useState('')
   567→
   568→  const toggle = (id: string) =>
   569→    setExpanded((s) => {
   570→      const next = new Set(s)
   571→      if (next.has(id)) next.delete(id)
   572→      else next.add(id)
   573→      return next
   574→    })
   575→
   576→  const expandAll = () => {
   577→    const next = new Set<string>()
   578→    for (const g of tree) {
   579→      next.add(`g-${g.group.id}`)
   580→      for (const t of g.tiers) next.add(`t-${g.group.id}-${t.tier.id}`)
   581→    }
   582→    setExpanded(next)
   583→  }
   584→  const collapseAll = () => setExpanded(new Set())
   585→
   586→  const filtered = useMemo(() => {
   587→    if (!query.trim()) return tree
   588→    const q = query.toLowerCase()
   589→    return tree
   590→      .map((g) => {
   591→        const gmatch =
   592→          g.group.name.zhHant.toLowerCase().includes(q) ||
   593→          g.group.name.en.toLowerCase().includes(q) ||
   594→          `g${g.group.id}`.includes(q)
   595→        const tiers = g.tiers
   596→          .map((t) => {
   597→            const tmatch =
   598→              t.tier.name.toLowerCase().includes(q) ||
   599→              t.tier.category.zhHant?.toLowerCase().includes(q) ||
   600→              t.tier.category.en.toLowerCase().includes(q) ||
   601→              `t${t.tier.id}`.includes(q)
   602→            const products = t.products.filter(
   603→              (p) =>
   604→                p.name.zhHant.toLowerCase().includes(q) ||
   605→                p.name.en.toLowerCase().includes(q) ||
   606→                `p${p.id}`.includes(q) ||
   607→                String(p.id).includes(q),
   608→            )
   609→            if (tmatch || products.length > 0) {
   610→              return { ...t, products: tmatch ? t.products : products }
   611→            }
   612→            return null
   613→          })
   614→          .filter((x): x is NonNullable<typeof x> => x != null)
   615→        if (gmatch || tiers.length > 0) {
   616→          return { ...g, tiers: gmatch ? g.tiers : tiers }
   617→        }
   618→        return null
   619→      })
   620→      .filter((x): x is NonNullable<typeof x> => x != null)
   621→  }, [tree, query])
   622→
   623→  // auto-expand matches while searching
   624→  const effectiveExpanded = useMemo(() => {
   625→    if (!query.trim()) return expanded
   626→    const next = new Set(expanded)
   627→    for (const g of filtered) {
   628→      next.add(`g-${g.group.id}`)
   629→      for (const t of g.tiers) next.add(`t-${g.group.id}-${t.tier.id}`)
   630→    }
   631→    return next
   632→  }, [expanded, filtered, query])
   633→
   634→  return (
   635→    <Card className="p-4">
   636→      <div className="mb-3 flex flex-wrap items-center gap-2">
   637→        <div className="relative min-w-[200px] flex-1">
   638→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   639→          <Input
   640→            value={query}
   641→            onChange={(e) => setQuery(e.target.value)}
   642→            placeholder={pick(STR.searchPh, lang)}
   643→            className="pl-8"
   644→          />
   645→        </div>
   646→        <Button size="sm" variant="outline" onClick={expandAll}>
   647→          <ChevronDown className="h-4 w-4" />
   648→          {pick(STR.expandAll, lang)}
   649→        </Button>
   650→        <Button size="sm" variant="outline" onClick={collapseAll}>
   651→          <ChevronRight className="h-4 w-4" />
   652→          {pick(STR.collapseAll, lang)}
   653→        </Button>
   654→      </div>
   655→
   656→      <div className={`max-h-[640px] overflow-y-auto pr-1 ${SCROLLBAR_CLS}`}>
   657→        {filtered.length === 0 ? (
   658→          <div className="py-12 text-center text-sm text-muted-foreground">
   659→            {lang === 'en' ? 'No matches found.' : '找不到符合的項目。'}
   660→          </div>
   661→        ) : (
   662→          <div className="space-y-1">
   663→            {filtered.map((g) => {
   664→              const gid = `g-${g.group.id}`
   665→              const gOpen = effectiveExpanded.has(gid)
   666→              const dot = `rgb(${Math.round(g.group.color.r * 255)},${Math.round(g.group.color.g * 255)},${Math.round(g.group.color.b * 255)})`
   667→              return (
   668→                <div key={gid}>
   669→                  <button
   670→                    onClick={() => toggle(gid)}
   671→                    className="flex w-full items-center gap-2 rounded-md border bg-card/60 px-3 py-2 text-left transition-colors hover:border-primary/50 hover:bg-accent"
   672→                  >
   673→                    {gOpen ? (
   674→                      <ChevronDown className="h-4 w-4 shrink-0 text-muted-foreground" />
   675→                    ) : (
   676→                      <ChevronRight className="h-4 w-4 shrink-0 text-muted-foreground" />
   677→                    )}
   678→                    <span className="inline-block h-3 w-3 shrink-0 rounded-full" style={{ background: dot }} />
   679→                    <span className="flex-1 truncate text-sm font-semibold">
   680→                      {groupIdNameFor(g.group.id, lang)}
   681→                    </span>
   682→                    <Badge variant="secondary" className="shrink-0">
   683→                      {g.tiers.length} {pick(STR.tiers, lang)}
   684→                    </Badge>
   685→                    <Badge variant="outline" className="shrink-0">
   686→                      {g.totalProducts} {pick(STR.products, lang)}
   687→                    </Badge>
   688→                  </button>
   689→                  {gOpen && (
   690→                    <div className="ml-3 mt-1 space-y-1 border-l pl-3">
   691→                      {g.tiers.map((t) => {
   692→                        const tid = `t-${g.group.id}-${t.tier.id}`
   693→                        const tOpen = effectiveExpanded.has(tid)
   694→                        return (
   695→                          <div key={tid}>
   696→                            <button
   697→                              onClick={() => toggle(tid)}
   698→                              className="flex w-full items-center gap-2 rounded-md px-3 py-1.5 text-left transition-colors hover:bg-accent"
   699→                            >
   700→                              {tOpen ? (
   701→                                <ChevronDown className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />
   702→                              ) : (
   703→                                <ChevronRight className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />
   704→                              )}
   705→                              <Layers className="h-3.5 w-3.5 shrink-0 text-sky-500" />
   706→                              <span className="flex-1 truncate text-sm">
   707→                                {tierIdNameFor(t.tier.id, lang)}
   708→                              </span>
   709→                              <span className="shrink-0 text-[10px] text-muted-foreground">
   710→                                ×{t.tier.inflation}
   711→                              </span>
   712→                              <Badge variant="secondary" className="shrink-0 text-[10px]">
   713→                                {t.products.length}
   714→                              </Badge>
   715→                            </button>
   716→                            {tOpen && (
   717→                              <div className="ml-3 mt-0.5 space-y-0.5 border-l pl-3">
   718→                                {t.products.map((p) => {
   719→                                  const isPremium = encyclopedia.premiumProducts.includes(p.id)
   720→                                  return (
   721→                                    <div
   722→                                      key={p.id}
   723→                                      className="flex items-center gap-2 rounded-md px-3 py-1 text-xs transition-colors hover:bg-accent/60"
   724→                                    >
   725→                                      <Package className="h-3 w-3 shrink-0 text-emerald-500" />
   726→                                      <code className="shrink-0 rounded bg-muted px-1 text-[10px]">
   727→                                        #{p.id}
   728→                                      </code>
   729→                                      <span className="flex-1 truncate">{productNameFor(p.id, lang)}</span>
   730→                                      <span className="shrink-0 text-[10px] text-muted-foreground">
   731→                                        ${p.basePricePerUnit.toFixed(2)}
   732→                                      </span>
   733→                                      {isPremium && (
   734→                                        <Badge className="shrink-0 bg-amber-500/20 text-amber-700 text-[10px] dark:text-amber-400">
   735→                                          <Crown className="h-2.5 w-2.5" />
   736→                                          {lang === 'en' ? 'Premium' : '頂級'}
   737→                                        </Badge>
   738→                                      )}
   739→                                    </div>
   740→                                  )
   741→                                })}
   742→                              </div>
   743→                            )}
   744→                          </div>
   745→                        )
   746→                      })}
   747→                    </div>
   748→                  )}
   749→                </div>
   750→              )
   751→            })}
   752→          </div>
   753→        )}
   754→      </div>
   755→    </Card>
   756→  )
   757→}
