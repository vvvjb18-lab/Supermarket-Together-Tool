    50→interface Bounds {
    51→  minX: number
    52→  minY: number
    53→  maxX: number
    54→  maxY: number
    55→  w: number
    56→  h: number
    57→}
    58→
    59→function computeBounds(nodes: SkillGraphNode[]): Bounds {
    60→  if (nodes.length === 0) return { minX: 0, minY: 0, maxX: 1, maxY: 1, w: 1, h: 1 }
    61→  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity
    62→  for (const n of nodes) {
    63→    if (n.x < minX) minX = n.x
    64→    if (n.y < minY) minY = n.y
    65→    if (n.x > maxX) maxX = n.x
    66→    if (n.y > maxY) maxY = n.y
    67→  }
    68→  const pad = 40
    69→  return { minX: minX - pad, minY: minY - pad, maxX: maxX + pad, maxY: maxY + pad, w: maxX - minX + pad * 2, h: maxY - minY + pad * 2 }
    70→}

   473→      if (s.perk != null) m.set(s.perk, s)
   474→    }
   475→    return m
   476→  }, [])
   477→
   478→  const rows = useMemo(() => {
   479→    const perks = skillGraph.nodes.filter((n) => n.type === 'perk')
   480→    let list = perks
   481→    if (filterPlaceholder === 'real') list = list.filter((n) => !n.is_placeholder)
   482→    else if (filterPlaceholder === 'placeholder') list = list.filter((n) => n.is_placeholder)
   483→    const q = query.trim().toLowerCase()
   484→    if (q) {
   485→      list = list.filter((n) =>
   486→        n.name_en.toLowerCase().includes(q) ||
   487→        (n.name_zhHant ?? '').toLowerCase().includes(q) ||
   488→        (n.skill_id ?? '').toLowerCase().includes(q) ||
   489→        (n.desc_en ?? '').toLowerCase().includes(q) ||
   490→        String(n.index).includes(q)
   491→      )
   492→    }
   493→    const sorted = [...list].sort((a, b) => {
   494→      let av: string | number, bv: string | number
   495→      if (sortKey === 'index') { av = a.index; bv = b.index }
   496→      else if (sortKey === 'name') { av = (a.name_zhHant || a.name_en).toLowerCase(); bv = (b.name_zhHant || b.name_en).toLowerCase() }
   497→      else if (sortKey === 'category') {
   498→        av = skillGraph.perk_to_category[a.id]?.category_name ?? ''
   499→        bv = skillGraph.perk_to_category[b.id]?.category_name ?? ''
   500→      }
   501→      else { av = a.is_placeholder ? 1 : 0; bv = b.is_placeholder ? 1 : 0 }
   502→      if (av < bv) return sortDir === 'asc' ? -1 : 1
   503→      if (av > bv) return sortDir === 'asc' ? 1 : -1
   504→      return 0
   505→    })
   506→    return sorted
   507→  }, [query, sortKey, sortDir, filterPlaceholder])
   508→
   509→  const toggleSort = (k: SortKey) => {
   510→    if (sortKey === k) setSortDir((d) => (d === 'asc' ? 'desc' : 'asc'))
   511→    else { setSortKey(k); setSortDir('asc') }
   512→  }
   513→
   514→  const nameFor = (n: SkillGraphNode): string => {
   515→    if (lang === 'en') return n.name_en || '—'
   516→    if (lang === 'zhHant') return n.name_zhHant || n.name_en || '—'
   517→    if (n.name_zhHant && n.name_en && n.name_zhHant !== n.name_en) return `${n.name_zhHant} / ${n.name_en}`
   518→    return n.name_zhHant || n.name_en || '—'
   519→  }
   520→
   521→  return (
   522→    <Card>
   523→      <CardHeader className="pb-2">
   524→        <CardTitle className="flex flex-wrap items-center justify-between gap-2 text-base">
   525→          <span className="flex items-center gap-2">
   526→            <Search className="h-4 w-4 text-emerald-500" />
   527→            44 技能完整總表
   528→          </span>
   529→          <div className="flex flex-wrap items-center gap-2">
   530→            <Badge variant="outline" className="text-[10px]">{rows.length} / 44</Badge>
   531→            {unlockedSet.size > 0 && (
   532→              <Badge variant="outline" className="text-[10px] bg-emerald-500/10 text-emerald-700 dark:text-emerald-300">
   533→                <Unlock className="mr-1 h-2.5 w-2.5" />{unlockedSet.size} 已解鎖
   534→              </Badge>
   535→            )}
   536→            <ConfidenceBadge
   537→              confidence="confirmed"
   538→              formula="perks.tsv + skill-graph.json"
   539→              note="perk_index 0-43 對應存檔 skill_unlocks 陣列索引"
   540→            />
   541→          </div>
   542→        </CardTitle>
   543→      </CardHeader>
   544→      <CardContent className="space-y-3">
   545→        {/* Filter bar */}
   546→        <div className="flex flex-wrap items-center gap-2">
   547→          <div className="relative flex-1 min-w-[200px]">
   548→            <Search className="absolute left-2 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   549→            <input
   550→              className="h-8 w-full rounded-md border bg-background pl-7 pr-3 text-xs"
   551→              placeholder="搜尋名稱 / skill_id / 描述…"
   552→              value={query}
   553→              onChange={(e) => setQuery(e.target.value)}
   554→            />
   555→          </div>
   556→          <div className="flex items-center gap-1">
   557→            {(['all', 'real', 'placeholder'] as const).map((f) => (
   558→              <Button
   559→                key={f}
   560→                variant={filterPlaceholder === f ? 'default' : 'outline'}
   561→                size="sm"
   562→                className="h-7 text-xs"
   563→                onClick={() => setFilterPlaceholder(f)}
   564→              >
   565→                {f === 'all' ? '全部' : f === 'real' ? '實技能' : '員工佔位'}
   566→              </Button>
   567→            ))}
   568→          </div>
   569→        </div>
   570→
   571→        {/* Table */}
   572→        <div className="max-h-[560px] overflow-auto scrollbar-thin rounded-md border">
   573→          <Table>
   574→            <TableHeader className="sticky top-0 bg-card z-10">
   575→              <TableRow>
   576→                <TableHead className="w-12 cursor-pointer text-right" onClick={() => toggleSort('index')}>
   577→                  # {sortKey === 'index' && (sortDir === 'asc' ? '↑' : '↓')}
   578→                </TableHead>
   579→                <TableHead className="min-w-[160px] cursor-pointer" onClick={() => toggleSort('name')}>
   580→                  名稱 {sortKey === 'name' && (sortDir === 'asc' ? '↑' : '↓')}
   581→                </TableHead>
   582→                <TableHead className="text-xs">skill_id</TableHead>
   583→                <TableHead className="min-w-[200px]">效果 / 描述</TableHead>
   584→                <TableHead className="cursor-pointer text-xs" onClick={() => toggleSort('category')}>
   585→                  最近分類 {sortKey === 'category' && (sortDir === 'asc' ? '↑' : '↓')}
   586→                </TableHead>
   587→                <TableHead className="cursor-pointer text-center" onClick={() => toggleSort('placeholder')}>
   588→                  類型 {sortKey === 'placeholder' && (sortDir === 'asc' ? '↑' : '↓')}
   589→                </TableHead>
   590→                <TableHead className="text-center">狀態</TableHead>
   591→                <TableHead className="text-right">座標</TableHead>
   592→              </TableRow>
   593→            </TableHeader>
   594→            <TableBody>
   595→              {rows.map((n) => {
   596→                const enc = encSkillByPerk.get(n.index)
   597→                const isUnlocked = unlockedSet.has(n.index)
   598→                const cat = skillGraph.perk_to_category[n.id]
   599→                return (
   600→                  <TableRow key={n.id} data-state={isUnlocked ? 'selected' : undefined}>
   601→                    <TableCell className="text-right font-mono text-xs text-muted-foreground">{n.index}</TableCell>
   602→                    <TableCell>
   603→                      <div className="font-medium text-xs">{nameFor(n)}</div>
   604→                      {n.name_en && n.name_zhHant && n.name_en !== n.name_zhHant && lang !== 'en' && (
   605→                        <div className="text-[10px] text-muted-foreground">{n.name_en}</div>
   606→                      )}
   607→                    </TableCell>
   608→                    <TableCell className="font-mono text-[10px] text-muted-foreground">{n.skill_id || '—'}</TableCell>
   609→                    <TableCell>
   610→                      <span className="text-xs leading-relaxed text-muted-foreground whitespace-normal">
   611→                        {n.desc_en || enc?.effect || '—'}
   612→                      </span>
   613→                    </TableCell>
   614→                    <TableCell>
   615→                      <span className="font-mono text-[10px] text-muted-foreground">
   616→                        {cat?.category_name ?? '—'}
   617→                      </span>
   618→                    </TableCell>
   619→                    <TableCell className="text-center">
   620→                      {n.is_placeholder ? (
   621→                        <Badge variant="outline" className="text-[9px] bg-amber-500/10 text-amber-700 dark:text-amber-300 border-amber-500/30">
   622→                          佔位
   623→                        </Badge>
   624→                      ) : (
   625→                        <Badge variant="outline" className="text-[9px] bg-emerald-500/10 text-emerald-700 dark:text-emerald-300 border-emerald-500/30">
   626→                          實技能
   627→                        </Badge>
   628→                      )}
   629→                    </TableCell>
   630→                    <TableCell className="text-center">
   631→                      {isUnlocked ? (
   632→                        <Badge variant="outline" className="text-[9px] bg-fuchsia-500/10 text-fuchsia-700 dark:text-fuchsia-300 border-fuchsia-500/30">
   633→                          <Unlock className="mr-1 h-2.5 w-2.5" />已解鎖
   634→                        </Badge>
   635→                      ) : (
   636→                        <span className="text-[10px] text-muted-foreground">—</span>
   637→                      )}
   638→                    </TableCell>
   639→                    <TableCell className="text-right font-mono text-[10px] text-muted-foreground">
   640→                      ({n.x.toFixed(0)}, {n.y.toFixed(0)})
   641→                    </TableCell>
   642→                  </TableRow>
   643→                )
   644→              })}
   645→            </TableBody>
   646→          </Table>
   647→        </div>
   648→
   649→        {/* Footer summary */}
   650→        <div className="grid grid-cols-2 gap-3 text-xs sm:grid-cols-4">
   651→          <div className="rounded-md border bg-card p-2">
   652→            <div className="text-[10px] text-muted-foreground">總 perk 數</div>
   653→            <div className="font-mono text-lg font-bold">44</div>
   654→          </div>
   655→          <div className="rounded-md border bg-card p-2">
   656→            <div className="text-[10px] text-muted-foreground">實技能</div>
   657→            <div className="font-mono text-lg font-bold text-emerald-600 dark:text-emerald-400">
   658→              {skillGraph.nodes.filter((n) => n.type === 'perk' && !n.is_placeholder).length}
   659→            </div>
   660→          </div>
   661→          <div className="rounded-md border bg-card p-2">
   662→            <div className="text-[10px] text-muted-foreground">員工佔位</div>
   663→            <div className="font-mono text-lg font-bold text-amber-600 dark:text-amber-400">
   664→              {skillGraph.nodes.filter((n) => n.type === 'perk' && n.is_placeholder).length}
   665→            </div>
   666→          </div>
   667→          <div className="rounded-md border bg-card p-2">
   668→            <div className="text-[10px] text-muted-foreground">總成本（全解鎖）</div>
   669→            <div className="font-mono text-lg font-bold">44,000 FP</div>
   670→          </div>
   671→        </div>
   672→      </CardContent>
   673→    </Card>
   674→  )
