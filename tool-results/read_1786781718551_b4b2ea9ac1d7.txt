    51→// ============================================================
    52→// Store Layout Analyzer
    53→// ------------------------------------------------------------
    54→// Top-down SVG map of all store props (save data if available,
    55→// else encyclopedia demo layout of 41 props). Each prop is
    56→// positioned by posX/posZ, rotated by angle, colored by
    57→// buildableId. Multiple highlight modes surface empty / low /
    58→// high-value / duplicated / anomaly shelves.
    59→// ============================================================
    60→
    61→type HighlightMode =
    62→  | 'none'
    63→  | 'empty'
    64→  | 'low'
    65→  | 'high-value'
    66→  | 'low-value'
    67→  | 'duplicated'
    68→  | 'missing-demand'
    69→  | 'negative'
    70→
    71→const BUILDABLE_PALETTE: Record<number, string> = {
    72→  0: '#94a3b8', // slate-400 — Placement Mode (default shelf)
    73→  1: '#10b981', // emerald-500 — Product Shelf
    74→  2: '#0ea5e9', // sky-500 — Basic Fridge
    75→  3: '#6366f1', // indigo-500 — Double Fridge
    76→}
    77→const FALLBACK_COLOR = '#a1a1aa'
    78→
    79→type SortKey = 'efficiency' | 'shelfValue' | 'totalUnits' | 'demandCoverage' | 'emptySlots' | 'distinctProducts'
    80→
    81→const HIGHLIGHT_LABELS: { value: HighlightMode; label: string; color: string }[] = [
    82→  { value: 'none', label: '無', color: '#a1a1aa' },
    83→  { value: 'empty', label: '空貨架', color: '#f43f5e' },
    84→  { value: 'low', label: '低庫存', color: '#f59e0b' },
    85→  { value: 'high-value', label: '高價值', color: '#10b981' },
    86→  { value: 'low-value', label: '低價值', color: '#71717a' },
    87→  { value: 'duplicated', label: '重複商品', color: '#a855f7' },
    88→  { value: 'missing-demand', label: '缺高需求', color: '#fb923c' },
    89→  { value: 'negative', label: '負庫存', color: '#e11d48' },
    90→]
    91→
    92→export function StoreLayout() {
    93→  const lang = useLang()
    94→  const snapshot = useSaveStore((s) => s.snapshot)
    95→  const room = useRoomStore((s) => s.room)
    96→  const assignShelf = useRoomStore((s) => s.assignShelf)
    97→
    98→  const layout: LayoutProp[] = useMemo(
    99→    () => snapshot?.storeLayout ?? ENC.storeLayout,
   100→    [snapshot],
   101→  )
   102→
   103→  const effResult = useMemo(() => computeShelfEfficiency(layout), [layout])
   104→  const efficiencies = effResult.value
   105→
   106→  // Precompute product demand for "missing-demand" highlight.
   107→  const demandCache = useMemo(() => {
   108→    const m = new Map<number, number>()
   109→    for (const p of ENC.products) {
   110→      m.set(p.id, computeDemandProxy(p.id, ENC.necessities, ENC.customerTypes).value)
   111→    }
   112→    return m
   113→  }, [])
   114→
   115→  const [highlight, setHighlight] = useState<HighlightMode>('none')
   116→  // selectedIdx and sortKey are used in the leaderboard + detail card below
   117→  const [selectedIdx, setSelectedIdx] = useState<number | null>(null)
   118→  const [sortKey, setSortKey] = useState<SortKey>('efficiency')
   119→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc')
   120→  const [assignPlayerId, setAssignPlayerId] = useState<string | undefined>(undefined)
   121→
   122→  // Map bounds (with padding) for the SVG viewBox.
   123→  const bounds = useMemo(() => {
   124→    if (layout.length === 0) return { minX: 0, minY: 0, w: 10, h: 10 }
   125→    const xs = layout.map((p) => p.posX)
   126→    const zs = layout.map((p) => p.posZ)
   127→    const minX = Math.min(...xs) - 1
   128→    const maxX = Math.max(...xs) + 1
   129→    const minZ = Math.min(...zs) - 1
   130→    const maxZ = Math.max(...zs) + 1
   131→    return { minX, minZ, w: maxX - minX, h: maxZ - minZ }
   132→  }, [layout])
   133→
   134→  // Highlight thresholds.
   135→  const thresholds = useMemo(() => {
   136→    if (efficiencies.length === 0) {
   137→      return { valueTop: 0, valueBottom: 0 }
   138→    }
   139→    const sorted = [...efficiencies].sort((a, b) => a.shelfValue - b.shelfValue)
   140→    const q = Math.max(1, Math.floor(efficiencies.length / 4))
   141→    return {
   142→      valueTop: sorted[sorted.length - q].shelfValue,
   143→      valueBottom: sorted[q - 1].shelfValue,
   144→    }
   145→  }, [efficiencies])
   146→
   147→  const getHighlight = (eff: PropEfficiency): { stroke: string; width: number; blink?: boolean } | null => {
   148→    if (highlight === 'none') return null
   149→    switch (highlight) {
   150→      case 'empty':
   151→        return eff.emptySlots > 0 ? { stroke: '#f43f5e', width: 0.08 } : null
   152→      case 'low':
   153→        return eff.totalUnits < 5 ? { stroke: '#f59e0b', width: 0.08 } : null
   154→      case 'high-value':
   155→        return eff.shelfValue >= thresholds.valueTop ? { stroke: '#10b981', width: 0.08 } : null
   156→      case 'low-value':
   157→        return eff.shelfValue <= thresholds.valueBottom ? { stroke: '#71717a', width: 0.08 } : null
   158→      case 'duplicated':
   159→        return eff.duplicatedProducts > 0 ? { stroke: '#a855f7', width: 0.08 } : null
   160→      case 'missing-demand': {
   161→        const prop = layout.find((p) => p.index === eff.propIndex)
   162→        if (!prop) return null
   163→        const maxDemand = Math.max(
   164→          0,
   165→          ...prop.inventory
   166→            .filter((i) => i.product >= 0 && i.count > 0)
   167→            .map((i) => demandCache.get(i.product) ?? 0),
   168→        )
   169→        return maxDemand < 0.002 ? { stroke: '#fb923c', width: 0.08 } : null
   170→      }
   171→      case 'negative':
   172→        return eff.negativeAnomalies > 0 ? { stroke: '#e11d48', width: 0.1, blink: true } : null
   173→    }
   174→    return null
   175→  }
   176→
   177→  // Sorted leaderboard rows.
   178→  const sortedEffs = useMemo(() => {
   179→    const arr = [...efficiencies]
   180→    arr.sort((a, b) => {
   181→      const av = a[sortKey]
   182→      const bv = b[sortKey]
   183→      return sortDir === 'asc' ? av - bv : bv - av
   184→    })
   185→    return arr
   186→  }, [efficiencies, sortKey, sortDir])
   187→
   188→  const selectedProp = selectedIdx != null ? layout.find((p) => p.index === selectedIdx) ?? null : null
   189→  const selectedEff = selectedIdx != null ? efficiencies.find((e) => e.propIndex === selectedIdx) ?? null : null
   190→
   191→  // Layout-wide aggregates for the report.
   192→  const aggregates = useMemo(() => {
   193→    const totalProps = layout.length
   194→    const totalUnits = efficiencies.reduce((s, e) => s + e.totalUnits, 0)
   195→    const totalShelfValue = efficiencies.reduce((s, e) => s + e.shelfValue, 0)
   196→    const totalEmptySlots = efficiencies.reduce((s, e) => s + e.emptySlots, 0)
   197→    const totalAnomalies = efficiencies.reduce((s, e) => s + e.negativeAnomalies, 0)
   198→    const totalDuplicated = efficiencies.reduce((s, e) => s + e.duplicatedProducts, 0)
   199→    const totalDemandCoverage = efficiencies.reduce((s, e) => s + e.demandCoverage, 0)
   200→    const demandCoveragePct = totalUnits > 0 ? Math.min(100, (totalDemandCoverage / totalUnits) * 1000) : 0
   201→
   202→    // Inventory by product (aggregate across props).
   203→    const byProduct = new Map<number, number>()
   204→    for (const prop of layout) {
   205→      for (const inv of prop.inventory) {
   206→        if (inv.product < 0 || inv.count <= 0) continue
   207→        byProduct.set(inv.product, (byProduct.get(inv.product) ?? 0) + inv.count)
   208→      }
   209→    }
   210→    const topProducts = [...byProduct.entries()]
   211→      .sort((a, b) => b[1] - a[1])
   212→      .slice(0, 20)
   213→      .map(([pid, count]) => ({ pid, count }))
   214→
   215→    // Inventory by group.
   216→    const byGroup = new Map<string, { count: number; value: number }>()
   217→    for (const prop of layout) {
   218→      for (const inv of prop.inventory) {
   219→        if (inv.product < 0 || inv.count <= 0) continue
   220→        const p = ENC.products.find((x) => x.id === inv.product)
   221→        if (!p) continue
   222→        const gname = groupIdNameFor(p.group, lang)
   223→        const cur = byGroup.get(gname) ?? { count: 0, value: 0 }
   224→        cur.count += inv.count
   225→        cur.value += inv.count * p.basePricePerUnit
   226→        byGroup.set(gname, cur)
   227→      }
   228→    }
   229→    const topGroups = [...byGroup.entries()]
   230→      .sort((a, b) => b[1].value - a[1].value)
   231→      .map(([name, v]) => ({ name, ...v }))
   232→
   233→    return {
   234→      totalProps,
   235→      totalUnits,
   236→      totalShelfValue,
   237→      totalEmptySlots,
   238→      totalAnomalies,
   239→      totalDuplicated,
   240→      demandCoveragePct,
   241→      topProducts,
   242→      topGroups,
   243→    }
   244→  }, [layout, efficiencies, lang])
   245→
   246→  // Recommended shelf swaps.
   247→  const swapRecs = useMemo(() => {
   248→    const recs: { from: number; to: number; reason: string }[] = []
   249→    const byEff = [...efficiencies].sort((a, b) => a.efficiency - b.efficiency)
   250→    // Bottom 5 props — suggest action.
   251→    for (const e of byEff.slice(0, 5)) {
   252→      if (e.emptySlots > 0) {
   253→        recs.push({
   254→          from: e.propIndex,
   255→          to: -1,
   256→          reason: `貨架 #${e.propIndex} 有 ${e.emptySlots} 個空格 — 從 storage 補上高 demand 商品`,
   257→        })
   258→      } else if (e.demandCoverage < 0.0001 && e.totalUnits > 0) {
   259→        recs.push({
   260→          from: e.propIndex,
   261→          to: -1,
   262→          reason: `貨架 #${e.propIndex} 商品 demand 偏低 (coverage=${fmt(e.demandCoverage, 5)}) — 考慮替換為高 demand 商品`,
   263→        })
   264→      }
   265→    }
   266→    // Duplicated products across props — suggest consolidation.
   267→    const productToProps = new Map<number, number[]>()
   268→    for (const prop of layout) {
   269→      for (const inv of prop.inventory) {
   270→        if (inv.product < 0 || inv.count <= 0) continue
   271→        const arr = productToProps.get(inv.product) ?? []
   272→        arr.push(prop.index)
   273→        productToProps.set(inv.product, arr)
   274→      }
   275→    }
   276→    let dupCount = 0
   277→    for (const [pid, props] of productToProps) {
   278→      if (props.length > 1 && dupCount < 5) {
   279→        recs.push({
   280→          from: props[0],
   281→          to: props[1],
   282→          reason: `商品 ${productNameFor(pid, lang)} (#${pid}) 同時出現在貨架 #${props[0]} 與 #${props[1]} — 考慮集中放置`,
   283→        })
   284→        dupCount++
   285→      }
   286→    }
   287→    return recs.slice(0, 8)
   288→  }, [layout, efficiencies, lang])
   289→
   290→  // Top 5 problematic shelves — quick scan card
   291→  const topProblematic = useMemo(() => {
   292→    return [...efficiencies]
   293→      .map((e) => {
   294→        const issues: string[] = []
   295→        if (e.negativeAnomalies > 0) issues.push(`負庫存 ×${e.negativeAnomalies}`)
   296→        if (e.totalUnits === 0) issues.push('完全空貨架')
   297→        if (e.emptySlots > 0 && e.totalUnits > 0) issues.push(`${e.emptySlots} 空格`)
   298→        if (e.duplicatedProducts > 0) issues.push(`${e.duplicatedProducts} 重複商品`)
   299→        if (e.demandCoverage < 0.0001 && e.totalUnits > 0) issues.push('低需求商品')
   300→        // weight: anomalies matter most, then empties, then duplicates, then low-demand
   301→        const score =
   302→          e.negativeAnomalies * 10 +
   303→          (e.totalUnits === 0 ? 5 : 0) +
   304→          e.emptySlots +
   305→          e.duplicatedProducts * 2 +
   306→          (e.demandCoverage < 0.0001 && e.totalUnits > 0 ? 1 : 0)
   307→        let recommendation = ''
   308→        if (e.negativeAnomalies > 0) recommendation = '檢查存檔資料，可能有資料異常'
   309→        else if (e.totalUnits === 0) recommendation = '從倉庫補上高 demand 商品'
   310→        else if (e.emptySlots > 0) recommendation = `補滿 ${e.emptySlots} 個空格`
   311→        else if (e.duplicatedProducts > 0) recommendation = '集中重複商品到單一貨架'
   312→        else if (e.demandCoverage < 0.0001) recommendation = '替換為高 demand 商品'
   313→        return { e, issues, recommendation, score }
   314→      })
   315→      .filter((x) => x.issues.length > 0)
   316→      .sort((a, b) => b.score - a.score)
   317→      .slice(0, 5)
   318→  }, [efficiencies])
   319→
   320→  const layoutSource = snapshot ? `save (${snapshot.source})` : 'encyclopedia (demo)'
   321→
   322→  return (
   323→    <div className="mx-auto max-w-[1600px] space-y-4 p-4">
   324→      <SectionHeader
   325→        title="店面平面圖分析"
   326→        description={`來源: ${layoutSource} · ${layout.length} props · ${aggregates.totalUnits} units`}
   327→        confidence={snapshot ? 'confirmed' : 'demo'}
   328→        formula="snapshot.storeLayout ?? encyclopedia.storeLayout"
   329→        note={snapshot ? '從存檔讀取之實際店面配置' : '未上傳存檔 — 顯示百科 demo 配置（41 props）'}
   330→        right={
   331→          <Badge variant="outline" className="text-xs">
   332→            <Layers className="mr-1 h-3 w-3" /> {layout.length} props
   333→          </Badge>
   334→        }
   335→      />
   336→
   337→      {/* Map card */}
   338→      <Card>
   339→        <CardHeader className="pb-2">
   340→          <div className="flex flex-wrap items-center justify-between gap-2">
   341→            <CardTitle className="flex items-center gap-2 text-base">
   342→              <MapIcon className="h-4 w-4 text-emerald-500" />
   343→              俯視地圖
   344→            </CardTitle>
   345→            <div className="flex flex-wrap items-center gap-2">
   346→              <span className="text-xs text-muted-foreground">高亮模式:</span>
   347→              <ToggleGroup
   348→                type="single"
   349→                value={highlight}
   350→                onValueChange={(v) => v && setHighlight(v as HighlightMode)}
   351→                variant="outline"
   352→                size="sm"
   353→                className="flex-wrap"
   354→              >
   355→                {HIGHLIGHT_LABELS.map((h) => (
   356→                  <ToggleGroupItem key={h.value} value={h.value} className="text-xs">
   357→                    <span
   358→                      className="mr-1 inline-block h-2 w-2 rounded-full"
   359→                      style={{ backgroundColor: h.color }}
   360→                    />
   361→                    {h.label}
   362→                  </ToggleGroupItem>
   363→                ))}
   364→              </ToggleGroup>
   365→            </div>
   366→          </div>
   367→        </CardHeader>
   368→        <CardContent>
   369→          <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   370→            {/* SVG map */}
   371→            <div className="lg:col-span-2">
   372→              <div className="overflow-x-auto scrollbar-thin">
   373→                <svg
   374→                  viewBox={`${bounds.minX} ${bounds.minZ} ${bounds.w} ${bounds.h}`}
   375→                  preserveAspectRatio="xMidYMid meet"
   376→                  className="w-full min-w-[480px] rounded-md border bg-muted/30"
   377→                  style={{ aspectRatio: `${bounds.w} / ${bounds.h}` }}
   378→                >
   379→                  {/* Grid lines */}
   380→                  <defs>
   381→                    <pattern id="grid" width="1" height="1" patternUnits="userSpaceOnUse">
   382→                      <path d="M 1 0 L 0 0 0 1" fill="none" stroke="currentColor" strokeWidth="0.02" className="text-muted-foreground/30" />
   383→                    </pattern>
   384→                  </defs>
   385→                  <rect
   386→                    x={bounds.minX}
   387→                    y={bounds.minZ}
   388→                    width={bounds.w}
   389→                    height={bounds.h}
   390→                    fill="url(#grid)"
   391→                  />
   392→
   393→                  {/* Props */}
   394→                  {layout.map((prop) => {
   395→                    const eff = efficiencies.find((e) => e.propIndex === prop.index)
   396→                    if (!eff) return null
   397→                    const color = BUILDABLE_PALETTE[prop.buildableId] ?? FALLBACK_COLOR
   398→                    const hl = getHighlight(eff)
   399→                    const assignment = room?.shelfAssignments[String(prop.index)]
   400→                    const assignedMember = assignment ? room?.members.find((m) => m.id === assignment) : null
   401→                    const isSelected = selectedIdx === prop.index
   402→                    const rectW = 0.6
   403→                    const rectH = 0.4
   404→                    return (
   405→                      <g
   406→                        key={prop.index}
   407→                        transform={`translate(${prop.posX}, ${prop.posZ}) rotate(${prop.angle})`}
   408→                        onClick={() => setSelectedIdx(prop.index)}
   409→                        style={{ cursor: 'pointer' }}
   410→                      >
   411→                        {/* Assigned player ring */}
   412→                        {assignedMember && (
   413→                          <rect
   414→                            x={-rectW / 2 - 0.08}
   415→                            y={-rectH / 2 - 0.08}
   416→                            width={rectW + 0.16}
   417→                            height={rectH + 0.16}
   418→                            fill="none"
   419→                            stroke={assignedMember.color}
   420→                            strokeWidth={0.06}
   421→                            rx={0.05}
   422→                          />
   423→                        )}
   424→                        {/* Prop body */}
   425→                        <rect
   426→                          x={-rectW / 2}
   427→                          y={-rectH / 2}
   428→                          width={rectW}
   429→                          height={rectH}
   430→                          fill={color}
   431→                          stroke={hl?.stroke ?? (isSelected ? '#0a0a0a' : '#00000033')}
   432→                          strokeWidth={hl?.width ?? (isSelected ? 0.08 : 0.02)}
   433→                          rx={0.04}
   434→                          className={hl?.blink ? 'animate-pulse' : ''}
   435→                        />
   436→                        {/* Inventory count label */}
   437→                        <text
   438→                          x={0}
   439→                          y={0}
   440→                          textAnchor="middle"
   441→                          dominantBaseline="central"
   442→                          fontSize={0.28}
   443→                          className="fill-white font-bold pointer-events-none"
   444→                          style={{ paintOrder: 'stroke', stroke: '#00000080', strokeWidth: 0.03 }}
   445→                        >
   446→                          {eff.totalUnits}
   447→                        </text>
   448→                        {/* Prop index (small, top-right) */}
   449→                        <text
   450→                          x={rectW / 2 - 0.04}
   451→                          y={-rectH / 2 + 0.04}
   452→                          textAnchor="end"
   453→                          dominantBaseline="hanging"
   454→                          fontSize={0.13}
   455→                          className="fill-white/80 pointer-events-none"
   456→                        >
   457→                          #{prop.index}
   458→                        </text>
   459→                      </g>
   460→                    )
   461→                  })}
   462→                </svg>
   463→              </div>
   464→
   465→              {/* Legend */}
   466→              <div className="mt-2 flex flex-wrap items-center gap-3 text-xs">
   467→                <span className="text-muted-foreground">Buildable 顏色:</span>
   468→                {[0, 1, 2, 3].map((id) => (
   469→                  <div key={id} className="flex items-center gap-1.5">
   470→                    <span
   471→                      className="inline-block h-3 w-3 rounded-sm"
   472→                      style={{ backgroundColor: BUILDABLE_PALETTE[id] ?? FALLBACK_COLOR }}
   473→                    />
   474→                    <span>{buildableIdNameFor(id, lang)}</span>
   475→                  </div>
   476→                ))}
   477→                <span className="ml-2 text-muted-foreground">數字 = 該貨架總庫存單位</span>
   478→              </div>
   479→            </div>
   480→
   481→            {/* Detail panel */}
   482→            <div>
   483→              {selectedProp && selectedEff ? (
   484→                <PropDetailCard
   485→                  prop={selectedProp}
   486→                  eff={selectedEff}
   487→                  demandCache={demandCache}
   488→                  lang={lang}
   489→                  onClose={() => setSelectedIdx(null)}
   490→                  room={room}
   491→                  onAssign={(playerId) => assignShelf(String(selectedProp.index), playerId)}
   492→                />
   493→              ) : (
   494→                <div className="flex h-full flex-col items-center justify-center rounded-md border border-dashed bg-muted/20 p-6 text-center text-sm text-muted-foreground">
   495→                  <MapIcon className="mb-2 h-8 w-8 opacity-50" />
   496→                  <div>點擊地圖上任一貨架查看詳細資訊</div>
   497→                  <div className="mt-1 text-xs">含庫存清單、效率、缺漏、負庫存異常</div>
   498→                </div>
   499→              )}
   500→            </div>
   501→          </div>
   502→        </CardContent>

   835→  label,
   836→  k,
   837→  sortKey,
   838→  sortDir,
   839→  onSort,
   840→}: {
   841→  label: string
   842→  k: SortKey

   845→  onSort: (k: SortKey, dir: 'asc' | 'desc') => void
   846→}) {
   847→  const active = sortKey === k

   849→    <TableHead className="text-right">
   850→      <button
   851→        className="inline-flex items-center gap-1 text-xs hover:text-foreground"

   855→        <ArrowUpDown className={`h-3 w-3 ${active ? 'opacity-100' : 'opacity-30'}`} />
   856→        {active && <span className="text-[9px]">{sortDir === 'desc' ? '↓' : '↑'}</span>}
   857→      </button>
   858→    </TableHead>
   859→  )
   860→}
