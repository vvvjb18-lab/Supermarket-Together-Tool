    51→    51→// ============================================================
    52→    52→// Store Layout Analyzer
    53→    53→// ------------------------------------------------------------
    54→    54→// Top-down SVG map of all store props (save data if available,
    55→    55→// else encyclopedia demo layout of 41 props). Each prop is
    56→    56→// positioned by posX/posZ, rotated by angle, colored by
    57→    57→// buildableId. Multiple highlight modes surface empty / low /
    58→    58→// high-value / duplicated / anomaly shelves.

    60→    60→
    61→    61→type HighlightMode =
    62→    62→  | 'none'
    63→    63→  | 'empty'
    64→    64→  | 'low'
    65→    65→  | 'high-value'
    66→    66→  | 'low-value'
    67→    67→  | 'duplicated'
    68→    68→  | 'missing-demand'
    69→    69→  | 'negative'
    70→    70→
    71→    71→const BUILDABLE_PALETTE: Record<number, string> = {
    72→    72→  0: '#94a3b8', // slate-400 — Placement Mode (default shelf)
    73→    73→  1: '#10b981', // emerald-500 — Product Shelf
    74→    74→  2: '#0ea5e9', // sky-500 — Basic Fridge
    75→    75→  3: '#6366f1', // indigo-500 — Double Fridge
    76→    76→}
    77→    77→const FALLBACK_COLOR = '#a1a1aa'
    78→    78→
    79→    79→type SortKey = 'efficiency' | 'shelfValue' | 'totalUnits' | 'demandCoverage' | 'emptySlots' | 'distinctProducts'
    80→    80→
    81→    81→const HIGHLIGHT_LABELS: { value: HighlightMode; label: string; color: string }[] = [
    82→    82→  { value: 'none', label: '無', color: '#a1a1aa' },
    83→    83→  { value: 'empty', label: '空貨架', color: '#f43f5e' },
    84→    84→  { value: 'low', label: '低庫存', color: '#f59e0b' },
    85→    85→  { value: 'high-value', label: '高價值', color: '#10b981' },
    86→    86→  { value: 'low-value', label: '低價值', color: '#71717a' },
    87→    87→  { value: 'duplicated', label: '重複商品', color: '#a855f7' },
    88→    88→  { value: 'missing-demand', label: '缺高需求', color: '#fb923c' },
    89→    89→  { value: 'negative', label: '負庫存', color: '#e11d48' },
    90→    90→]
    91→    91→
    92→    92→export function StoreLayout() {
    93→    93→  const lang = useLang()
    94→    94→  const snapshot = useSaveStore((s) => s.snapshot)
    95→    95→  const room = useRoomStore((s) => s.room)
    96→    96→  const assignShelf = useRoomStore((s) => s.assignShelf)
    97→    97→
    98→    98→  const layout: LayoutProp[] = useMemo(
    99→    99→    () => snapshot?.storeLayout ?? ENC.storeLayout,
   100→   100→    [snapshot],
   101→   101→  )
   102→   102→
   103→   103→  const effResult = useMemo(() => computeShelfEfficiency(layout), [layout])
   104→   104→  const efficiencies = effResult.value
   105→   105→
   106→   106→  // Precompute product demand for "missing-demand" highlight.
   107→   107→  const demandCache = useMemo(() => {
   108→   108→    const m = new Map<number, number>()
   109→   109→    for (const p of ENC.products) {
   110→   110→      m.set(p.id, computeDemandProxy(p.id, ENC.necessities, ENC.customerTypes).value)
   111→   111→    }
   112→   112→    return m
   113→   113→  }, [])
   114→   114→
   115→   115→  const [highlight, setHighlight] = useState<HighlightMode>('none')
   116→   116→  // selectedIdx and sortKey are used in the leaderboard + detail card below
   117→   117→  const [selectedIdx, setSelectedIdx] = useState<number | null>(null)
   118→   118→  const [sortKey, setSortKey] = useState<SortKey>('efficiency')
   119→   119→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc')
   120→   120→  const [assignPlayerId, setAssignPlayerId] = useState<string | undefined>(undefined)
   121→   121→
   122→   122→  // Map bounds (with padding) for the SVG viewBox.
   123→   123→  const bounds = useMemo(() => {
   124→   124→    if (layout.length === 0) return { minX: 0, minY: 0, w: 10, h: 10 }
   125→   125→    const xs = layout.map((p) => p.posX)
   126→   126→    const zs = layout.map((p) => p.posZ)
   127→   127→    const minX = Math.min(...xs) - 1
   128→   128→    const maxX = Math.max(...xs) + 1
   129→   129→    const minZ = Math.min(...zs) - 1
   130→   130→    const maxZ = Math.max(...zs) + 1
   131→   131→    return { minX, minZ, w: maxX - minX, h: maxZ - minZ }
   132→   132→  }, [layout])
   133→   133→
   134→   134→  // Highlight thresholds.
   135→   135→  const thresholds = useMemo(() => {
   136→   136→    if (efficiencies.length === 0) {
   137→   137→      return { valueTop: 0, valueBottom: 0 }
   138→   138→    }
   139→   139→    const sorted = [...efficiencies].sort((a, b) => a.shelfValue - b.shelfValue)
   140→   140→    const q = Math.max(1, Math.floor(efficiencies.length / 4))
   141→   141→    return {
   142→   142→      valueTop: sorted[sorted.length - q].shelfValue,
   143→   143→      valueBottom: sorted[q - 1].shelfValue,
   144→   144→    }
   145→   145→  }, [efficiencies])
   146→   146→
   147→   147→  const getHighlight = (eff: PropEfficiency): { stroke: string; width: number; blink?: boolean } | null => {
   148→   148→    if (highlight === 'none') return null
   149→   149→    switch (highlight) {
   150→   150→      case 'empty':
   151→   151→        return eff.emptySlots > 0 ? { stroke: '#f43f5e', width: 0.08 } : null
   152→   152→      case 'low':
   153→   153→        return eff.totalUnits < 5 ? { stroke: '#f59e0b', width: 0.08 } : null
   154→   154→      case 'high-value':
   155→   155→        return eff.shelfValue >= thresholds.valueTop ? { stroke: '#10b981', width: 0.08 } : null
   156→   156→      case 'low-value':
   157→   157→        return eff.shelfValue <= thresholds.valueBottom ? { stroke: '#71717a', width: 0.08 } : null
   158→   158→      case 'duplicated':
   159→   159→        return eff.duplicatedProducts > 0 ? { stroke: '#a855f7', width: 0.08 } : null
   160→   160→      case 'missing-demand': {
   161→   161→        const prop = layout.find((p) => p.index === eff.propIndex)
   162→   162→        if (!prop) return null
   163→   163→        const maxDemand = Math.max(
   164→   164→          0,
   165→   165→          ...prop.inventory
   166→   166→            .filter((i) => i.product >= 0 && i.count > 0)
   167→   167→            .map((i) => demandCache.get(i.product) ?? 0),
   168→   168→        )
   169→   169→        return maxDemand < 0.002 ? { stroke: '#fb923c', width: 0.08 } : null
   170→   170→      }
   171→   171→      case 'negative':
   172→   172→        return eff.negativeAnomalies > 0 ? { stroke: '#e11d48', width: 0.1, blink: true } : null
   173→   173→    }
   174→   174→    return null
   175→   175→  }
   176→   176→
   177→   177→  // Sorted leaderboard rows.
   178→   178→  const sortedEffs = useMemo(() => {
   179→   179→    const arr = [...efficiencies]
   180→   180→    arr.sort((a, b) => {
   181→   181→      const av = a[sortKey]
   182→   182→      const bv = b[sortKey]
   183→   183→      return sortDir === 'asc' ? av - bv : bv - av
   184→   184→    })
   185→   185→    return arr
   186→   186→  }, [efficiencies, sortKey, sortDir])
   187→   187→
   188→   188→  const selectedProp = selectedIdx != null ? layout.find((p) => p.index === selectedIdx) ?? null : null
   189→   189→  const selectedEff = selectedIdx != null ? efficiencies.find((e) => e.propIndex === selectedIdx) ?? null : null
   190→   190→
   191→   191→  // Layout-wide aggregates for the report.
   192→   192→  const aggregates = useMemo(() => {
   193→   193→    const totalProps = layout.length
   194→   194→    const totalUnits = efficiencies.reduce((s, e) => s + e.totalUnits, 0)
   195→   195→    const totalShelfValue = efficiencies.reduce((s, e) => s + e.shelfValue, 0)
   196→   196→    const totalEmptySlots = efficiencies.reduce((s, e) => s + e.emptySlots, 0)
   197→   197→    const totalAnomalies = efficiencies.reduce((s, e) => s + e.negativeAnomalies, 0)
   198→   198→    const totalDuplicated = efficiencies.reduce((s, e) => s + e.duplicatedProducts, 0)
   199→   199→    const totalDemandCoverage = efficiencies.reduce((s, e) => s + e.demandCoverage, 0)
   200→   200→    const demandCoveragePct = totalUnits > 0 ? Math.min(100, (totalDemandCoverage / totalUnits) * 1000) : 0
   201→   201→
   202→   202→    // Inventory by product (aggregate across props).
   203→   203→    const byProduct = new Map<number, number>()
   204→   204→    for (const prop of layout) {
   205→   205→      for (const inv of prop.inventory) {
   206→   206→        if (inv.product < 0 || inv.count <= 0) continue
   207→   207→        byProduct.set(inv.product, (byProduct.get(inv.product) ?? 0) + inv.count)
   208→   208→      }
   209→   209→    }
   210→   210→    const topProducts = [...byProduct.entries()]
   211→   211→      .sort((a, b) => b[1] - a[1])
   212→   212→      .slice(0, 20)
   213→   213→      .map(([pid, count]) => ({ pid, count }))
   214→   214→
   215→   215→    // Inventory by group.
   216→   216→    const byGroup = new Map<string, { count: number; value: number }>()
   217→   217→    for (const prop of layout) {
   218→   218→      for (const inv of prop.inventory) {
   219→   219→        if (inv.product < 0 || inv.count <= 0) continue
   220→   220→        const p = ENC.products.find((x) => x.id === inv.product)
   221→   221→        if (!p) continue
   222→   222→        const gname = groupIdNameFor(p.group, lang)
   223→   223→        const cur = byGroup.get(gname) ?? { count: 0, value: 0 }
   224→   224→        cur.count += inv.count
   225→   225→        cur.value += inv.count * p.basePricePerUnit
   226→   226→        byGroup.set(gname, cur)
   227→   227→      }
   228→   228→    }
   229→   229→    const topGroups = [...byGroup.entries()]
   230→   230→      .sort((a, b) => b[1].value - a[1].value)
   231→   231→      .map(([name, v]) => ({ name, ...v }))
   232→   232→
   233→   233→    return {
   234→   234→      totalProps,
   235→   235→      totalUnits,
   236→   236→      totalShelfValue,
   237→   237→      totalEmptySlots,
   238→   238→      totalAnomalies,
   239→   239→      totalDuplicated,
   240→   240→      demandCoveragePct,
   241→   241→      topProducts,
   242→   242→      topGroups,
   243→   243→    }
   244→   244→  }, [layout, efficiencies, lang])
   245→   245→
   246→   246→  // Recommended shelf swaps.
   247→   247→  const swapRecs = useMemo(() => {
   248→   248→    const recs: { from: number; to: number; reason: string }[] = []
   249→   249→    const byEff = [...efficiencies].sort((a, b) => a.efficiency - b.efficiency)
   250→   250→    // Bottom 5 props — suggest action.
   251→   251→    for (const e of byEff.slice(0, 5)) {
   252→   252→      if (e.emptySlots > 0) {
   253→   253→        recs.push({
   254→   254→          from: e.propIndex,
   255→   255→          to: -1,
   256→   256→          reason: `貨架 #${e.propIndex} 有 ${e.emptySlots} 個空格 — 從 storage 補上高 demand 商品`,
   257→   257→        })
   258→   258→      } else if (e.demandCoverage < 0.0001 && e.totalUnits > 0) {
   259→   259→        recs.push({
   260→   260→          from: e.propIndex,
   261→   261→          to: -1,
   262→   262→          reason: `貨架 #${e.propIndex} 商品 demand 偏低 (coverage=${fmt(e.demandCoverage, 5)}) — 考慮替換為高 demand 商品`,
   263→   263→        })
   264→   264→      }
   265→   265→    }
   266→   266→    // Duplicated products across props — suggest consolidation.
   267→   267→    const productToProps = new Map<number, number[]>()
   268→   268→    for (const prop of layout) {
   269→   269→      for (const inv of prop.inventory) {
   270→   270→        if (inv.product < 0 || inv.count <= 0) continue
   271→   271→        const arr = productToProps.get(inv.product) ?? []
   272→   272→        arr.push(prop.index)
   273→   273→        productToProps.set(inv.product, arr)
   274→   274→      }
   275→   275→    }
   276→   276→    let dupCount = 0
   277→   277→    for (const [pid, props] of productToProps) {
   278→   278→      if (props.length > 1 && dupCount < 5) {
   279→   279→        recs.push({
   280→   280→          from: props[0],
   281→   281→          to: props[1],
   282→   282→          reason: `商品 ${productNameFor(pid, lang)} (#${pid}) 同時出現在貨架 #${props[0]} 與 #${props[1]} — 考慮集中放置`,
   283→   283→        })
   284→   284→        dupCount++
   285→   285→      }
   286→   286→    }
   287→   287→    return recs.slice(0, 8)
   288→   288→  }, [layout, efficiencies, lang])
   289→   289→
   290→   290→  // Top 5 problematic shelves — quick scan card
   291→   291→  const topProblematic = useMemo(() => {
   292→   292→    return [...efficiencies]
   293→   293→      .map((e) => {
   294→   294→        const issues: string[] = []
   295→   295→        if (e.negativeAnomalies > 0) issues.push(`負庫存 ×${e.negativeAnomalies}`)
   296→   296→        if (e.totalUnits === 0) issues.push('完全空貨架')
   297→   297→        if (e.emptySlots > 0 && e.totalUnits > 0) issues.push(`${e.emptySlots} 空格`)
   298→   298→        if (e.duplicatedProducts > 0) issues.push(`${e.duplicatedProducts} 重複商品`)
   299→   299→        if (e.demandCoverage < 0.0001 && e.totalUnits > 0) issues.push('低需求商品')
   300→   300→        // weight: anomalies matter most, then empties, then duplicates, then low-demand
   301→   301→        const score =
   302→   302→          e.negativeAnomalies * 10 +
   303→   303→          (e.totalUnits === 0 ? 5 : 0) +
   304→   304→          e.emptySlots +
   305→   305→          e.duplicatedProducts * 2 +
   306→   306→          (e.demandCoverage < 0.0001 && e.totalUnits > 0 ? 1 : 0)
   307→   307→        let recommendation = ''
   308→   308→        if (e.negativeAnomalies > 0) recommendation = '檢查存檔資料，可能有資料異常'
   309→   309→        else if (e.totalUnits === 0) recommendation = '從倉庫補上高 demand 商品'
   310→   310→        else if (e.emptySlots > 0) recommendation = `補滿 ${e.emptySlots} 個空格`
   311→   311→        else if (e.duplicatedProducts > 0) recommendation = '集中重複商品到單一貨架'
   312→   312→        else if (e.demandCoverage < 0.0001) recommendation = '替換為高 demand 商品'
   313→   313→        return { e, issues, recommendation, score }
   314→   314→      })
   315→   315→      .filter((x) => x.issues.length > 0)
   316→   316→      .sort((a, b) => b.score - a.score)
   317→   317→      .slice(0, 5)
   318→   318→  }, [efficiencies])
   319→   319→
   320→   320→  const layoutSource = snapshot ? `save (${snapshot.source})` : 'encyclopedia (demo)'
   321→   321→
   322→   322→  return (
   323→   323→    <div className="mx-auto max-w-[1600px] space-y-4 p-4">
   324→   324→      <SectionHeader
   325→   325→        title="店面平面圖分析"
   326→   326→        description={`來源: ${layoutSource} · ${layout.length} props · ${aggregates.totalUnits} units`}
   327→   327→        confidence={snapshot ? 'confirmed' : 'demo'}
   328→   328→        formula="snapshot.storeLayout ?? encyclopedia.storeLayout"
   329→   329→        note={snapshot ? '從存檔讀取之實際店面配置' : '未上傳存檔 — 顯示百科 demo 配置（41 props）'}
   330→   330→        right={
   331→   331→          <Badge variant="outline" className="text-xs">
   332→   332→            <Layers className="mr-1 h-3 w-3" /> {layout.length} props
   333→   333→          </Badge>
   334→   334→        }
   335→   335→      />
   336→   336→
   337→   337→      {/* Map card */}
   338→   338→      <Card>
   339→   339→        <CardHeader className="pb-2">
   340→   340→          <div className="flex flex-wrap items-center justify-between gap-2">
   341→   341→            <CardTitle className="flex items-center gap-2 text-base">
   342→   342→              <MapIcon className="h-4 w-4 text-emerald-500" />
   343→   343→              俯視地圖
   344→   344→            </CardTitle>
   345→   345→            <div className="flex flex-wrap items-center gap-2">
   346→   346→              <span className="text-xs text-muted-foreground">高亮模式:</span>
   347→   347→              <ToggleGroup
   348→   348→                type="single"
   349→   349→                value={highlight}
   350→   350→                onValueChange={(v) => v && setHighlight(v as HighlightMode)}
   351→   351→                variant="outline"
   352→   352→                size="sm"
   353→   353→                className="flex-wrap"
   354→   354→              >
   355→   355→                {HIGHLIGHT_LABELS.map((h) => (
   356→   356→                  <ToggleGroupItem key={h.value} value={h.value} className="text-xs">
   357→   357→                    <span
   358→   358→                      className="mr-1 inline-block h-2 w-2 rounded-full"
   359→   359→                      style={{ backgroundColor: h.color }}
   360→   360→                    />
   361→   361→                    {h.label}
   362→   362→                  </ToggleGroupItem>
   363→   363→                ))}
   364→   364→              </ToggleGroup>
   365→   365→            </div>
   366→   366→          </div>
   367→   367→        </CardHeader>
   368→   368→        <CardContent>
   369→   369→          <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   370→   370→            {/* SVG map */}
   371→   371→            <div className="lg:col-span-2">
   372→   372→              <div className="overflow-x-auto scrollbar-thin">
   373→   373→                <svg
   374→   374→                  viewBox={`${bounds.minX} ${bounds.minZ} ${bounds.w} ${bounds.h}`}
   375→   375→                  preserveAspectRatio="xMidYMid meet"
   376→   376→                  className="w-full min-w-[480px] rounded-md border bg-muted/30"
   377→   377→                  style={{ aspectRatio: `${bounds.w} / ${bounds.h}` }}
   378→   378→                >
   379→   379→                  {/* Grid lines */}
   380→   380→                  <defs>
   381→   381→                    <pattern id="grid" width="1" height="1" patternUnits="userSpaceOnUse">
   382→   382→                      <path d="M 1 0 L 0 0 0 1" fill="none" stroke="currentColor" strokeWidth="0.02" className="text-muted-foreground/30" />
   383→   383→                    </pattern>
   384→   384→                  </defs>
   385→   385→                  <rect
   386→   386→                    x={bounds.minX}
   387→   387→                    y={bounds.minZ}
   388→   388→                    width={bounds.w}
   389→   389→                    height={bounds.h}
   390→   390→                    fill="url(#grid)"
   391→   391→                  />
   392→   392→
   393→   393→                  {/* Props */}
   394→   394→                  {layout.map((prop) => {
   395→   395→                    const eff = efficiencies.find((e) => e.propIndex === prop.index)
   396→   396→                    if (!eff) return null
   397→   397→                    const color = BUILDABLE_PALETTE[prop.buildableId] ?? FALLBACK_COLOR
   398→   398→                    const hl = getHighlight(eff)
   399→   399→                    const assignment = room?.shelfAssignments[String(prop.index)]
   400→   400→                    const assignedMember = assignment ? room?.members.find((m) => m.id === assignment) : null
   401→   401→                    const isSelected = selectedIdx === prop.index
   402→   402→                    const rectW = 0.6
   403→   403→                    const rectH = 0.4
   404→   404→                    return (
   405→   405→                      <g
   406→   406→                        key={prop.index}
   407→   407→                        transform={`translate(${prop.posX}, ${prop.posZ}) rotate(${prop.angle})`}
   408→   408→                        onClick={() => setSelectedIdx(prop.index)}
   409→   409→                        style={{ cursor: 'pointer' }}
   410→   410→                      >
   411→   411→                        {/* Assigned player ring */}
   412→   412→                        {assignedMember && (
   413→   413→                          <rect
   414→   414→                            x={-rectW / 2 - 0.08}
   415→   415→                            y={-rectH / 2 - 0.08}
   416→   416→                            width={rectW + 0.16}
